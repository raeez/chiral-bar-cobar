\chapter{Bar--cobar reconstruction and the Koszul--Morita square}
\label{ch:bar-cobar}

\section{Cobar and the universal twisting morphism}

Let $C$ be a coaugmented conilpotent coassociative coalgebra in
$\FactD(X)$ with coaugmentation coideal $\bar C$. Its cobar is
\[
  \mathrm{Cobar}_X(C) = \bigl(T(s^{-1}\bar C),\ d_C + d_\Delta\bigr),
\]
with $d_\Delta$ the derivation induced by the reduced coproduct. The
universal twisting morphism $\tau_A\colon \Barperp_X(A)\to A$ is
projection to one-letter words followed by desuspension.

\section{The nilcomplete locus and the positive-weight proof}

\begin{theorem}[Internal associative Koszul duality]
\label{thm:koszul-internal}
\StatusProved{} Let $\mathcal V$ be a stable presentable symmetric
monoidal $\infty$-category whose tensor product preserves colimits
separately. Bar and cobar restrict to inverse equivalences
\[
  \Alg^{\mathrm{aug}}_{E_1}(\mathcal V)^{\mathrm{nilcomp}}
  \;\simeq\;
  \Coalg^{\mathrm{coaug}}_{E_1}(\mathcal V)^{\mathrm{conilcomp}} .
\]
Applied to $\mathcal V=\FactD(X)$ this is ordered chiral bar--cobar
duality.
\end{theorem}

\begin{proof}
The associative operad is reduced and Koszul self-dual up to suspension,
so the bar--cobar adjunction identifies unit and counit after
nilcompletion and conilcompletion. Heuts' maximality theorem
\cite{Heu24} identifies these as the largest full subcategories on which
both unit and counit are equivalences.
\end{proof}

\begin{theorem}[Positive reconstruction]
\label{thm:positive-reconstruction}
\StatusProved{} Let $A=\unit\oplus\bigoplus_{w>0}A_w$ be connected and
complete, with each weight admitting finitely many decompositions into
positive weights. Then the counit
\[
  \mathrm{Cobar}_X\,\Barperp_X(A)\longrightarrow A
\]
is a filtered quasi-isomorphism. For a conilpotent weight-complete
coalgebra with the same finite-decomposition property, the unit
$C\to\Barperp_X\mathrm{Cobar}_X(C)$ is a filtered quasi-isomorphism.
\end{theorem}

\begin{proof}
Fix total weight $w$; finitely many bar words contribute. Filter
$\mathrm{Cobar}\,\Barperp(A)$ by the number of bars surviving the cobar
multiplication. The bar of a free word has the augmented simplex as
indexing complex, and the extra degeneracy contracts every subdivided
word onto the single unsplit word, so the associated-graded counit is a
quasi-isomorphism in each weight. Completeness and exhaustiveness promote
this to the filtered statement. The coalgebra case is dual, using the
finite coradical filtration in each weight.
\end{proof}

Theorem~\ref{thm:positive-reconstruction} is elementary and independent of
\cite{Heu24}; it is the form used in every calculation of
Chapter~\ref{ch:enveloping-audit}.

\section{The augmentation boundary}

\begin{definition}[Koszul dual at the augmentation]
\label{def:koszul-dual}
$A^! := \RHom_A(\unit,\unit)$, the derived endomorphism algebra of the
augmentation boundary condition, with $E_1$ structure from composition.
The bar resolution gives $A^!\simeq\RHom_{\FactD(X)}(\Barperp_X(A),\unit)$
whenever the bar terms are dualisable or the continuous internal Hom is
used.
\end{definition}

\begin{theorem}[Double centralizer]
\label{thm:double-centralizer}
\StatusConditional{} Let $A$ be connective and augmented, suppose the
derived augmentation-completion functor is computed by the cobar
totalization of the Amitsur complex of $A\to\unit$, and suppose $\unit$ is
compact in the completed module category. Then
\[
  A^{!!} = \RHom_{A^!}(\unit,\unit) \;\simeq\; A^{\wedge}_\varepsilon ,
\]
the derived augmentation completion of $A$. On the nilcomplete locus this
agrees with $A$.
\end{theorem}

\begin{proof}
The bar resolution identifies the thick subcategory generated by $\unit$
with perfect $A^!$-modules; endomorphisms of the corresponding Yoneda
generator are the double centralizer. Completion along $\ker\varepsilon$
is the universal algebra acting identically on the
augmentation-generated subcategory, and the cobar totalization computes
that completion.
\end{proof}

Verdier duality is one comparison to the universal bar resolution among
several, not the definition of $A^!$. Quadratic duality, PBW, and
curvature are the others.

\begin{theorem}[Quadratic recognition]
\label{thm:quadratic}
\StatusProved{} Let $V$ be dualisable, $A=T(V)/(R)$ with $R\hookrightarrow
V\otimes^! V$, and $A^{q!}=T(V^*[-1])/(R^\perp)$. If the weight
components are dualisable and the diagonal Koszul complex is exact in
positive homological degree, the inclusion of the quadratic dual coalgebra
into $\Barperp_X(A)$ is a quasi-isomorphism and $A^!\simeq A^{q!}$.
\end{theorem}

\begin{proof}
Filter the bar by tensor weight; its linear strand is the quadratic
Koszul complex. Exactness in positive degree collapses the spectral
sequence onto the quadratic dual coalgebra, and continuous duality
converts that coalgebra into the stated algebra.
\end{proof}

\begin{theorem}[Curved PBW duality]
\label{thm:curved-pbw}
\StatusProved{} Let a filtered relation carry a linear and a constant
term, $r-\alpha(r)-\beta(r)$ for $r\in R$, with $\alpha\colon R\to V$ and
$\beta\colon R\to\unit$. If the Braverman--Gaitsgory conditions
\cite{BG96} hold for $A_{\alpha,\beta}$, its associated graded is
$T(V)/(R)$ and its Koszul dual is the curved dg algebra
$\bigl(A^{q!}, d_\alpha, h_\beta\bigr)$ with $d_\alpha^2=[h_\beta,-]$.
\end{theorem}

\begin{remark}[Relative bases are part of the statement]
\label{rem:relative}
Not every algebra of interest is augmented over $\unit$. A quiver algebra
requires a relative bar construction over its algebra of vertex
idempotents; a matrix algebra and a Weyl algebra admit no augmentation at
all and require relative, Rees, or curved replacements. The choice of
base, the dualisability of the generators, and the completion used are
indispensable hypotheses of each theorem.
\end{remark}

\section{The Koszul--Morita square}

\begin{theorem}[Two-sided transform]
\label{thm:koszul-morita}
\StatusConditional{} Under the hypotheses of
Theorem~\ref{thm:double-centralizer}, the two-sided Koszul transform
carries the diagonal bimodule to the diagonal bicomodule, the relative
tensor product to the derived cotensor product, and Hochschild theory to
coHochschild theory. The endomorphisms of the identity module functor are
\[
  \End(\id_{\mathrm{Mod}_A}) \;\simeq\; \RHom_{A^e}(A,A) =: Z(A),
\]
carrying the universal internal $E_2$ structure acting centrally on $A$.
\end{theorem}

\begin{proof}
The two-sided bar resolution is a bimodule resolution of the diagonal, and
the transform is the induced functor on bimodules; it is monoidal for the
relative tensor product by the same resolution, whence cotensor on the
dual side. The identification of $\End(\id)$ with $\RHom_{A^e}(A,A)$ is
the internal form of the standard computation of the centre of a module
category, and the $E_2$ structure is the Deligne-type brace structure on
Hochschild cochains.
\end{proof}

\begin{remark}[Centre is not bulk]
\label{rem:centre-not-bulk}
$Z(A)$ is terminal among algebraic central actions on the boundary
algebra. It is the universal closed sector internal to that boundary, not
a decoupled physical bulk. Promoting it requires a supplied bulk
factorisation algebra together with an open--closed comparison
$\mathcal O^{\mathrm{phys}}_{\mathrm{bulk}}(A)\to Z(A)$ proved to be an
equivalence; see No-go~\ref{nogo:bulk}.
\end{remark}
