\chapter{Suspensions, orientations, and low-arity signs}\label{app:signs}

\section{Cohomological suspension}
For homogeneous \(a\), \(|sa|=|a|+1\).  The Koszul rule for exchanging homogeneous factors \(u,v\) is
\[
 u\otimes v\longmapsto(-1)^{|u||v|}v\otimes u.
\]
The binary bar coderivation is fixed by
\[
 b_2(sa\otimes sb)=(-1)^{|a|}s(ab).
\]
On a word, moving this degree-one coderivation past the preceding suspended letters gives
\[
 b_2[a_1|\cdots|a_r]
 =\sum_{i=1}^{r-1}
 (-1)^{\sigma_i}
 [a_1|\cdots|a_ia_{i+1}|\cdots|a_r],
\]
\[
 \sigma_i=\sum_{j<i}(|a_j|+1)+|a_i|.
\]
For degree-zero letters the signs alternate.  Any convention differing by a global suspension conjugation produces an isomorphic complex; mixing conventions within one calculation does not.

\section{Four-letter bar word}
For ungraded letters and the alternating convention,
\[
 b[a|b|c|d]
 =[ab|c|d]-[a|bc|d]+[a|b|cd].
\]
Applying \(b\) again gives
\begin{align*}
 b^2[a|b|c|d]
 ={}&[(ab)c|d]-[ab|cd]\\
 &-[a(bc)|d]+[a|b(cd)]\\
 &+[ab|cd]-[a|(bc)d].
\end{align*}
The middle terms cancel, and associativity cancels the remaining two pairs.  This is the cellular boundary of the two-dimensional associahedron in its simplest word model.

\section{Two edge colours}
For a bicoloured graph \(G\), choose
\[
 \operatorname{or}(G)=\det E_X(G)\otimes\det E_\perp(G).
\]
Contracting an \(X\)-edge removes a basis vector from the first determinant; contracting a \(\perp\)-edge removes one from the second.  Reversing the contraction order exchanges two odd normal directions and contributes \(-1\).  Hence
\[
 d_Xd_\perp+d_\perp d_X=0.
\]
The same convention controls the two loop operators in the master equation.

\section{Arnold relation and KZ curvature}
For three points,
\[
 \omega_{ij}=d\log(z_i-z_j)
\]
satisfy
\[
 \omega_{12}\wedge\omega_{23}
 +\omega_{23}\wedge\omega_{31}
 +\omega_{31}\wedge\omega_{12}=0.
\]
The arity-three KZ curvature is
\begin{align*}
 F_\nabla={}&\kappa^2\bigl(
 [\Omega_{12},\Omega_{13}]\omega_{12}\wedge\omega_{13}
 +[\Omega_{12},\Omega_{23}]\omega_{12}\wedge\omega_{23}\\
 &\hspace{5em}+[\Omega_{13},\Omega_{23}]\omega_{13}\wedge\omega_{23}
 \bigr).
\end{align*}
The infinitesimal braid relation rewrites the three commutators as one common commutator with the coefficients in Arnold's relation, so \(F_\nabla=0\).

\section{Rational Yang--Baxter expansion}
For \(R_{ij}(u)=1-\hbar P_{ij}/u\), expand both sides of Yang--Baxter.  The order-\(\hbar\) terms agree trivially.  The order-\(\hbar^2\) identity reduces, after a common denominator, to the permutation identities
\[
 P_{12}P_{13}=P_{23}P_{12},
 \quad
 P_{13}P_{23}=P_{12}P_{13},
 \quad
 P_{12}P_{23}=P_{13}P_{12}.
\]
The cubic term agrees because both sides represent the same longest permutation.

\section{The \(A_2\) Hall relation at \(q=2\)}
In the basis \(([X],[Y])\),
\[
 e_1^2e_2=(3,3),
 \qquad
 e_1e_2e_1=(3,1),
 \qquad
 e_2e_1^2=(3,0).
\]
Therefore
\[
 (3,3)-3(3,1)+2(3,0)=(0,0).
\]
The symbolic proof in \cref{thm:hall-serre} holds for arbitrary \(q\); the finite check catches coefficient transpositions.

\section{Stable trace Jacobi test}
For labels \((a,b)\) with \(0\le a,b\le3\) and \(a+b>0\), the script evaluates all ordered triples under
\[
 [(a,b),(c,d)]=(ad-bc)(a+c-1,b+d-1).
\]
All \(15^3=3375\) Jacobi sums vanish exactly.  This is a finite sanity check of the symbolic Poisson proof.

\chapter{Generators-and-relations templates}\label{app:templates}

\section{Quadratic transverse presentation}
Given coefficient generators \(V\in\FactD(X)\) and relations \(R\to V\tensorpt V\), the strict quadratic algebra is
\[
 A=T_{\tensorpt}(V)/(R).
\]
The derived presentation is the homotopy pushout
\[
\begin{tikzcd}
 \Free_{\Eone}(R)\arrow[r,shift left=0.4ex,"\partial_0"]
 \arrow[r,shift right=0.4ex,"\partial_1"']
 \arrow[d]
 &\Free_{\Eone}(V)\arrow[d]\\
 \one\arrow[r]&A_{\mathrm{der}}.
\end{tikzcd}
\]
A cofibrant replacement adds syzygy cells whenever relations among relations occur.

\section{Associative chiral presentation}
For formal-disc generators \(M\) and ordered field relations \(R_X\), replace \(\Free_{\Eone}\) by \(\Free_{\Assch}\).  A relation can be a finite-pole identity
\[
 (z-w)^N\left(Y(a,z)Y(b,w)-\sum_i f_i(z-w)Y(b_i,w)Y(a_i,z)\right)=0.
\]
A quantum relation includes the \(\mathcal S\)-matrix coefficients.  The derived quotient remembers a null-homotopy of this field identity and its compatibility with nested substitutions.

\section{Doubly associative presentation}
Use the two-coloured operad \(\mathcal P\).  The data consist of
\[
 (V;R_\perp,R_X,R_{\mathrm{mix}}),
\]
where \(R_{\mathrm{mix}}\) contains the rectangular locality square and any braided permutation between middle factors.  Resolving the presentation automatically adds:
\begin{itemize}
\item transverse overlap cells;
\item chiral collision overlap cells;
\item mixed prisms comparing the two reduction orders;
\item higher cells filling the resulting spheres.
\end{itemize}
This is the promised model-category implementation of homotopically complete generators and relations.

\section{Relative presentation}
For a semisimple base \(S\), use free \(S\)-bimodule algebras and relative tensor products throughout.  The quiver template is
\[
 A=T_S(M)/(R),
 \qquad
 \Barop_S(A)=S\otimes_A^{\mathbb L}S.
\]
A coloured chiral quiver replaces every \(e_jMe_i\) by a coefficient object and imposes the factorisation compatibility on every source-target component.
