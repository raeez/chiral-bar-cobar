\section{4d/2d correspondence algebras}
\label{sec:4d2d}

\subsection{The 4d/2d correspondence (BLLPRR)}

\begin{construction}[4d/2d correspondence (BLLPRR)]\label{constr:4d2d}
Starting from a 4d $\mathcal{N} = 2$ superconformal field theory $\mathcal{T}_{4d}$:
\begin{enumerate}[label=(\roman*)]
\item Select the \emph{Schur supercharge} $\mathbb{Q} = Q_1^- + \widetilde{S}^{1\dot{-}}$, a specific nilpotent supercharge in the $\mathcal{N}=2$ superconformal algebra ($\mathbb{Q}^2 = 0$).
\item Pass to the cohomology $H^*(\mathbb{Q})$ of local operators inserted at a fixed plane $\mathbb{R}^2 \subset \mathbb{R}^4$.
\item The resulting algebra $\mathcal{A}[\mathcal{T}_{4d}] = H^*(\mathbb{Q}, \text{local ops})$ carries the structure of a 2d vertex algebra (chiral algebra).
\end{enumerate}

This construction, due to Beem--Lemos--Liendo--Peelaers--Rastelli--van~Rees~\cite{beem-et-al}, produces vertex algebras from 4d SCFTs.
It is distinct from the $\Omega$-background/Nekrasov partition-function
approach used in the AGT correspondence below.
\end{construction}

\begin{theorem}[4d/2d is $\Einf$-chiral; \ClaimStatusProvedElsewhere{} \cite{beem-et-al}]\label{thm:4d2d-einf}
The 2d chiral algebra $\mathcal{A}[\mathcal{T}_{4d}]$ obtained from the 4d/2d
correspondence is an $\Einf$-chiral algebra (vertex algebra).

The OPE of the resulting chiral algebra is generically \emph{non-commutative}: for instance, the 4d/2d image of $SU(2)$ $\mathcal{N}=4$ SYM is the small $\mathcal{N}=4$ superconformal vertex algebra at $c = -9$, which contains $\widehat{\mathfrak{sl}}_2$ at level $k = -3/2$ (an admissible, non-integral level; determined by $c = 6k$) as its R-symmetry current subalgebra.  The $\Einf$ structure arises from the full vertex algebra axioms (locality, associativity of OPE), not from commutativity.
\end{theorem}

\begin{example}[Class $\mathcal{S}$]\label{ex:class-s}
For class $\mathcal{S}$ theories of type $A_{n-1}$ on a Riemann surface $C$:
\[
\mathcal{A}[\mathcal{T}_{A_{n-1}}(C)] = \mathcal{W}^{-n}(\mathfrak{sl}_n)
\]
the W-algebra associated to $\mathfrak{sl}_n$ at the critical level $k = -n = -h^\vee$. At this special level the Sugawara construction degenerates and $\mathcal{W}^{-h^\vee}(\mathfrak{sl}_n)$ coincides with the Feigin--Frenkel center $\mathfrak{z}(\widehat{\mathfrak{g}})$, a commutative vertex algebra (see Feigin--Frenkel \cite{FF} for the center isomorphism and Arakawa \cite{Ara15} for related rationality results). The dependence on the Riemann surface $C$ and its pants decomposition appears at the level of conformal blocks and modules, not the algebra itself.
\end{example}


\section{Non-commutative Chern--Simons theory}
\label{sec:nc-chern-simons}
\noindent\emph{Status split.} All claims in this section are
\ClaimStatusConjectured; the purpose is to state a precise target
compatible with the proved bar-cobar formalism.

\subsection{Chern--Simons as source of chiral algebras}

\begin{construction}[CS boundary algebra]\label{constr:cs-boundary}
3d Chern--Simons theory with gauge group $G$ at level $k$ on $M^3 = \Sigma \times \bR_+$ 
produces:
\begin{enumerate}[label=(\roman*)]
\item On the boundary $\Sigma$: WZW model at level $k$.
\item The chiral algebra is the affine Kac--Moody algebra $\hat{\mathfrak{g}}_k$.
\end{enumerate}
\end{construction}

\begin{conjecture}[Non-commutative CS; \ClaimStatusConjectured]\label{conj:nc-cs}
Noncommutative Chern--Simons theory on $\bR^3_\theta$ produces
$\Eone$-chiral algebras:
\begin{enumerate}[label=(\roman*)]
\item The boundary theory is a non-commutative WZW model.
\item The OPE is R-twisted with $R$ depending on the non-commutativity parameter $\theta$.
\item At $\theta = 0$, we recover the standard $\Einf$-chiral Kac--Moody.
\end{enumerate}
\end{conjecture}

\begin{remark}[Scope]
Conjecture~\ref{conj:nc-cs} requires input from non-commutative gauge theory:
the construction of a consistent non-commutative WZW model on $\bR^3_\theta$
and the identification of its boundary algebra with an $\Eone$-chiral structure
depend on string-theoretic results (Seiberg--Witten \cite{SW99}) that lie outside
the scope of this monograph.  The mathematical content --- that $R$-twisted OPE
yields $\Eone$-chiral algebras interpolating to $\Einf$ at $\theta = 0$ --- is
consistent with the framework of this monograph but its verification requires non-commutative
deformation quantization of the Chern--Simons functional.  Even if this
boundary construction were established, it would still stop short of the
filtered H-level bulk objects needed for $W_\infty$-type or holographic
duals.  The theorematic standard-tower completed M-level packages are
already separated out.  The next live layer is MC4: bulk realization with
finite-quotient control.  The local tests are now explicit:
RTT-level preservation on the Yangian side and spin-triangular
higher-spin ideals on the $W_\infty$ side.  The next step after those
preservation theorems is equally explicit: extract the actual
coefficient families
$\mathsf{K}^{\mathrm{line}}_{a,b}(N)$ and
$\mathsf{C}^{\mathrm{res}}_{s,t;u;m,n}(N)$, compare them mode by mode
with the finite RTT and principal Drinfeld--Sokolov stages, and close
the finite-detection packets $\Delta_{a,0}(N)$ and $\mathcal{I}_N$.
Only after that comparison
package is in place does the downstream MC5 BV/BRST or holographic
identification become a well-posed target.  Concretely, the next layer
is the proof of
$\mathsf{K}^{\mathrm{line}}_{a,b}(N)=\mathsf{K}^{\mathrm{RTT}}_{a,b}(N)$
and
$\mathsf{C}^{\mathrm{res}}_{s,t;u;m,n}(N)
=\mathsf{C}^{\mathrm{DS}}_{s,t;u;m,n}(N)$.

\emph{Homotopy template} (Convention~\ref{conv:hms-levels}): Type~I (existence).
The conjecture asserts the existence of $\Eone$-chiral algebras from non-commutative Chern--Simons theory, with the non-commutativity parameter $\theta$ controlling the $R$-twist.
(Contributing to Conjecture~\ref{conj:master-bv-brst}.)
\end{remark}


\subsection{\texorpdfstring{Noncommutative geometry and associative / topological $\mathsf{E}_1$ structures}{Noncommutative geometry and associative / topological E1 structures}}

\begin{construction}[NC torus as associative deformation]
\label{constr:nc-torus-e1}
\index{noncommutative torus!topological $E_1$ structure@topological $\mathsf{E}_1$ structure}
The noncommutative torus $T^2_\theta$ ($\theta \in \mathbb{R} \setminus \mathbb{Q}$)
carries a natural associative deformation, equivalently a topological
$\Eone$-algebra structure.  The smooth algebra
$C^\infty(T^2_\theta)$ is generated by unitaries $U, V$ subject to $VU = e^{2\pi i \theta} UV$,
and the Moyal star product
\[
(f \star_\theta g)(x,y) = \sum_{m,n} \hat{f}(m,n)\,\hat{g}(m',n')\,
e^{i\pi\theta(mn'-m'n)}\,e^{2\pi i((m+m')x + (n+n')y)}
\]
is associative but noncommutative, yielding a topological
$\Eone$-structure.  What is \emph{not} constructed here is a
Beilinson--Drinfeld $\Eone$-chiral algebra on a curve.  The object at
hand is the associative deformation of the Poisson structure
$\{x,y\} = \theta$ on $T^2$.
\end{construction}

\begin{conjecture}[Koszul duality as Morita equivalence; \ClaimStatusConjectured]
\label{conj:koszul-morita}
\index{Morita equivalence!Koszul duality interpretation}
$\Eone$-Koszul duality for noncommutative tori exchanges the deformation parameter:
\[
\mathcal{A}_\theta^! \;\cong\; \mathcal{A}_{1/\theta}.
\]
This coincides with Rieffel's Morita equivalence $T^2_\theta \sim_{\mathrm{Morita}} T^2_{1/\theta}$
for irrational $\theta$, reinterpreted as $\Eone$-Koszul duality.
The bar complex $\barB(\mathcal{A}_\theta)$ computes the cyclic homology
$\mathrm{HC}_*(\mathcal{A}_\theta)$, which by Connes' theorem equals the de~Rham
cohomology $H^*_{\mathrm{dR}}(T^2)$ --- independent of $\theta$ (topological invariance
of cyclic homology for smooth deformations).
\end{conjecture}

\begin{remark}[Scope]
Conjecture~\ref{conj:koszul-morita} combines two established results ---
Rieffel's Morita equivalence (Rieffel, 1981) and Connes' computation of
$\mathrm{HC}_*(A_\theta)$ (Connes, 1985) --- with the unproved identification of
Morita equivalence as $\Eone$-Koszul duality.  The latter requires
constructing the relevant topological $\Eone$ bar complex for
$\mathcal{A}_\theta$ and verifying that its cohomology reproduces the
Koszul dual at $1/\theta$; this lies outside the algebraic-curve
setting of this monograph.  A curve-chiral realization would be an
additional structure, not something assumed here.

\emph{Homotopy template} (Convention~\ref{conv:hms-levels}): Type~II (equivalence).
The conjecture treats topological $\Eone$-Koszul duality for
noncommutative tori as the algebraic shadow of Rieffel's Morita equivalence
$T^2_\theta \sim T^2_{1/\theta}$.
(Contributing to Conjecture~\ref{conj:master-bv-brst}.)
\end{remark}


\section{Gauge theory and D-branes}
\label{sec:gauge-d-branes}
\noindent\emph{Status split.} Statements here are conjectural
physics-to-algebra identifications; only the abstract module-Koszul
machinery they invoke is proved in Part~I.

\subsection{D-brane vertex algebras}

\begin{construction}[Open string VOA]\label{constr:open-string-voa}
Consider open strings ending on D-branes in type IIB string theory:
\begin{enumerate}[label=(\roman*)]
\item The worldsheet is a disk $D$ with boundary conditions.
\item The boundary CFT gives rise to a chiral algebra.
\item For D-branes in a Calabi--Yau, the algebra depends on the D-brane configuration.
\end{enumerate}
\end{construction}

\begin{conjecture}[D-brane algebras are $\Eone$; \ClaimStatusConjectured]\label{conj:dbrane-e1}
Open string vertex algebras on non-trivial D-brane configurations are generically 
$\Eone$-chiral algebras:
\begin{enumerate}[label=(\roman*)]
\item The Chan--Paton factors introduce non-commutativity.
\item The OPE has a matrix structure: $\Phi^{ij}(z) \Psi^{k\ell}(w) \sim \delta^{jk} \cdots$.
\item Commutativity of the OPE fails due to the matrix ordering (the OPE is no longer symmetric under exchange of operators, yielding an $\Eone$- rather than $\Einf$-structure).
\end{enumerate}
\end{conjecture}

\begin{remark}[Scope]
Conjecture~\ref{conj:dbrane-e1} requires string-theoretic input: the construction
of open string vertex algebras from D-brane configurations in type~IIB string
theory, including the identification of Chan--Paton factors with matrix-valued
OPE and the precise mechanism by which boundary conditions break
skew-symmetry.  The mathematical consequence --- that matrix-valued OPE
structures produce $\Eone$-chiral algebras --- is compatible with
Definition~\ref{def:chiral-ass-operad} of the $\Eone$-chiral operad, but
its derivation from first principles requires open string field theory
(Zwiebach \cite{Zwi93}), which is beyond the scope of this monograph.
This chapter therefore records boundary/module-side targets only: the
theorematic standard-tower completed M-level packages are not the issue
here.  Any
corresponding bulk dual objects remain downstream in the same
dependency order as elsewhere in the book: first an MC4 filtered
H-level model recovering the proved finite stages, then the MC5
physics-side comparison to that model.

\emph{Homotopy template} (Convention~\ref{conv:hms-levels}): Type~VII (physics-dictionary).
The conjecture treats D-brane vertex algebras as a string-theoretic source for
$\Eone$-chiral algebras, with Chan--Paton factors supplying the
non-commutative shadow that breaks the $\Einf$ structure.
(Contributing to Conjecture~\ref{conj:master-bv-brst}.)
\end{remark}


\subsection{\texorpdfstring{D-brane categories from $\Eone$-module Koszul duality}{D-brane categories from E1-module Koszul duality}}

\begin{conjecture}[HMS from $\Eone$-module Koszul duality; \ClaimStatusConjectured]
\label{conj:hms-koszul}
\index{homological mirror symmetry!Koszul duality interpretation}
\index{D-brane!module Koszul duality}
In string theory, D-branes on a target space $X$ form the derived category
$D^b(\mathrm{Coh}(X))$ (B-branes) or the Fukaya category $\mathrm{Fuk}(X)$
(A-branes).  For a chiral algebra $\mathcal{A}$ on a curve $\Sigma$, the
module category $\mathrm{Mod}(\mathcal{A})$ is the mathematical incarnation
of the category of branes.

Under $\Eone$-Koszul duality, the module bar-cobar functor
$\Phi\colon \mathrm{Mod}(\mathcal{A}) \xrightarrow{\sim}
\mathrm{CoMod}(\mathcal{A}^!)$
of Theorem~\ref{thm:e1-module-koszul-duality} exchanges module and comodule
categories.  When $X$ is a Calabi--Yau surface and $\mathcal{A}$ is the chiral
algebra of the sigma model on $X$, this functor should recover homological
mirror symmetry (Kontsevich):
\[
D^b\bigl(\mathrm{Coh}(X)\bigr)
\;\simeq\;
D^b\bigl(\mathrm{Fuk}(X^\vee)\bigr)
\]
for a mirror pair $(X, X^\vee)$.  In this interpretation, the bar-cobar
adjunction is conjecturally interpreted as the passage from B-branes to A-branes: the bar construction
$\barB(\mathcal{A})$ computes the Ext-algebra of the diagonal brane,
and the cobar construction recovers the endomorphism algebra of the dual
collection on $X^\vee$.
\end{conjecture}

\begin{remark}[Scope]
Conjecture~\ref{conj:hms-koszul} is a physics-motivated reinterpretation
of homological mirror symmetry (Kontsevich \cite{Kon94}) through the lens
of $\Eone$-module Koszul duality.  The identification of the chiral algebra
module category with a brane category requires string-theoretic input
(boundary CFT, open string field theory), and the passage from the abstract
module bar-cobar functor $\Phi$ to the geometric equivalence of derived
categories requires additional analysis (e.g., generation of the derived
category by a tilting object compatible with the Koszul dual).  The
mathematical content of Theorem~\ref{thm:e1-module-koszul-duality} stands
independently; only the physical interpretation as HMS is conjectural.
As elsewhere in the repo's frontier order, the theorematic content here
is the module/comodule bar-cobar machinery itself; any geometric brane
realization is part of the downstream comparison layer, not a missing
finite-type algebraic input.

\emph{Homotopy template} (Convention~\ref{conv:hms-levels}): Type~VII (physics-dictionary).
The conjecture interprets the $\Eone$-module bar-cobar functor as the passage from B-branes to A-branes, recovering homological mirror symmetry from Koszul duality.
(Contributing to Conjecture~\ref{conj:master-bv-brst}.)
\end{remark}


\section{AGT correspondence connections}
\label{sec:agt}
\noindent\emph{Status split.} Classical AGT statements are imported as
\ClaimStatusProvedElsewhere, while the $q$-lift and the deformation of
the boundary chiral algebra from local/commutative to
nonlocal/associative remain \ClaimStatusConjectured.

\begin{theorem}[AGT correspondence; \ClaimStatusProvedElsewhere{} \cite{AGT09}]\label{thm:agt}
(Alday--Gaiotto--Tachikawa) For 4d $\mathcal{N} = 2$ $SU(2)$ gauge theory with 
$N_f = 4$ fundamental hypermultiplets:
\[
Z_{\mathrm{Nekrasov}}(\epsilon_1, \epsilon_2, a; q) = 
\langle V_{\alpha_1}(z_1) \cdots V_{\alpha_4}(z_4) \rangle_{\mathrm{Liouville}}
\]
where:
\begin{itemize}
\item LHS: Nekrasov partition function
\item RHS: 4-point conformal block in Liouville CFT
\item $c = 1 + 6Q^2$, $Q = b + b^{-1}$, $b^2 = \epsilon_1/\epsilon_2$ (dimensionless ratio of $\Omega$-background parameters)
\end{itemize}
\end{theorem}

\begin{corollary}[Virasoro from gauge theory; \ClaimStatusProvedElsewhere{} \cite{AGT09}]\label{cor:virasoro-gauge}
The Virasoro algebra (an $\Einf$-chiral algebra) arises from the gauge theory 
computation. The central charge $c$ is determined by the $\Omega$-background parameters.
\end{corollary}


\begin{conjecture}[$q$-AGT; \ClaimStatusConjectured]\label{conj:q-agt}
The 5d lift of AGT (adding a circle) gives:
\[
Z_{\mathrm{5d Nekrasov}}(q, t) = \text{(conformal blocks of } \mathcal{W}_{q,t})
\]
where $\mathcal{W}_{q,t}$ is the quantum W-algebra, an $\Eone$-chiral algebra.
\end{conjecture}

\begin{remark}[Scope]
Conjecture~\ref{conj:q-agt} requires the 5d lift of the AGT correspondence,
which involves the Nekrasov partition function of 5d $\mathcal{N}=1$
gauge theory compactified on $S^1$.  The identification of $\mathcal{W}_{q,t}$
as an $\Eone$-chiral algebra is mathematically well-defined (Frenkel--Reshetikhin),
but the equality with 5d gauge-theoretic partition functions is a physics
conjecture (Awata--Yamada \cite{AY10}, Nieri et al.)\ that goes beyond the
algebraic and geometric methods of this monograph.  The correct target category is $\Eone$-chiral algebras, but the
gauge-theoretic derivation lies outside the scope of this monograph.

\emph{Homotopy template} (Convention~\ref{conv:hms-levels}): Type~VII (physics-dictionary).
The conjecture treats conformal blocks of the quantum W-algebra
$\mathcal{W}_{q,t}$ as boundary-side algebraic shadows for 5d Nekrasov
partition functions, with the boundary chiral algebra deformed from local/commutative to
nonlocal/associative.
(Contributing to Conjecture~\ref{conj:master-bv-brst}.)
\end{remark}

\begin{remark}[From $\Einf$ to $\Eone$]
The dimensional lift $4d \to 5d \to 6d$ may deform the boundary chiral
algebra from local/commutative to nonlocal/associative: 4d produces
Virasoro ($\Einf$), 5d produces $q$-Virasoro ($\Eone$), and 6d
produces elliptic Virasoro.  This is a deformation inside
curve-chiral geometry, not an identification with the topological
little-disks ladder.
\end{remark}
