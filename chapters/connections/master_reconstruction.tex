\chapter[Master Reconstruction and Formation Theorem]%
{Master Reconstruction and Formation Theorem: the open Beilinson fork on a tangential
log curve}
\label{ch:master-reconstruction}

\begin{abstract}
On a tangential log curve $(X,D,\tau)$ the open quadrant of
$(\mathrm{Open}\cup\mathrm{CY})\times(\mathrm{Cat}\cup\mathrm{Chain})$
is organized by six typed objects and a fork of maps.  A factorization compact
generator $b$ of the stable presentable category $\mathsf C$ produces
the chart algebra $A_b=\RHom_{\mathsf C}(b,b)$.  Factorization Morita
reconstructs $\mathsf C$ from this pointed chart.  Enhanced
associative bar--cobar duality reconstructs $A_b$ from the full bar
coalgebra under the Ran and factorization-closure packages.

Three further constructions begin with the chart.  Chiral Hochschild
cochains form the chart center.  The quadratic--Verdier
comparison forms $A_b^!$ and its perfect label category.  Intrinsic
lines are the colimit-preserving factorization endofunctors of
$\mathsf C$.  The regular bimodule forms the pointed brane datum; under
$H_Z$ its derived endomorphisms agree with the chart center.  A full connected
category-valued modular functor restricts to its genus-zero balanced
braided line datum; its Hodge and partition classes are scalar
projections.

The Master Reconstruction Theorem records the two chart-side
reconstructions, the three further constructions, and the named
comparison problems among their targets.  A point of $\KSDual$ is a
homotopy-fixed triple $(A_b,\varphi_b,H_b)$ for the Verdier--Koszul
involution.  Vertical comparison functors connect
the Open nodes to their Vol~II and Vol~III companions under the
hypothesis package displayed at each node.
\end{abstract}

\section{The primitive open datum}
\label{sec:mr-seven-data}
%===========================================================

The open quadrant carries one structure with named components, all defined
before any choice of presentation.

\begin{definition}[Mixed Ran and open-sector factorization category;
\ClaimStatusDefinitional]
\label{def:mr-mixed-ran-factorization-category}
\index{mixed Ran space}
\index{factorization dg-category!open sector}
Let \((X,D,\tau)\) be a tangential log curve.  Write
\(\widetilde X_D\) for the real-oriented blowup of \(X\) at the points
of \(D\), and write
\[
 I_p:=S^1_p\setminus\{\tau_p\}\cong\mathbb R
 \qquad (p\in D)
\]
for the boundary interval obtained by deleting the tangential
basepoint.  For a finite set \(I\) of closed insertions and a tuple
\(\mathbf m=(m_p)_{p\in D}\) of boundary multiplicities, set
\[
\Conf^{\mathrm{oc}}_{I,\mathbf m}(X,D,\tau)
:=
\Conf_I(X\setminus D)
\times
\prod_{p\in D}\Conf^{<}_{m_p}(I_p).
\]
The \emph{mixed Ran space} is the colimit
\[
\Ran^{\mathrm{oc}}(X,D,\tau)
:=
\varinjlim_{(I,\mathbf m)}
\Conf^{\mathrm{oc}}_{I,\mathbf m}(X,D,\tau)/\Sigma_I .
\]
An \emph{open-sector factorization dg-category} on
\((X,D,\tau)\) is a constructible dg-cosheaf of dg-categories
\(\cC\) on \(\Ran^{\mathrm{oc}}(X,D,\tau)\) equipped with:
\begin{enumerate}[label=\textup{(\roman*)}]
\item factorization equivalences for decompositions into disjoint
sub-configurations;
\item holomorphic locality and meromorphic collision extension in the
closed variables;
\item local constancy on each boundary interval \(I_p\), giving the
\(E_1\)-direction;
\item a datum: a specified local mixed-collision model --- a
two-coloured collision structure on the boundary--bulk corestrictions
whose component operations are those of the local chiral Swiss-cheese
category (Theorem~\ref{thm:chiral-deligne-tamarkin}); the choice of
model is part of the hypothesis package of every statement that
consumes it; and
\item clutching compatibility for restoring the punctured boundary
circles and gluing boundary strata.
\end{enumerate}
This is the category-valued factorization package used in this
volume when we write ``factorization dg-category on \((X,D,\tau)\)''.
Francis--Gaitsgory factorization algebras supply the closed
\(D=\varnothing\) algebra-valued case.  The definition above supplies
the category-valued open-sector case.
\end{definition}

\begin{definition}[Factorization compact generator;
\ClaimStatusDefinitional]
\label{def:mr-factorization-compact-generator}
Let $U\mapsto\mathsf C(U)$ be the stable presentable enhancement of an
open-sector factorization category on
$\Ran^{\mathrm{oc}}(X,D,\tau)$.  Its corestriction functors are exact,
colimit preserving, and compact preserving.  A \emph{factorization
compact generator} is a
homotopy-coherent cocartesian section
$b=(b_U,\beta_f,\beta_{U,V})$ such that:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $b_U$ is a compact generator of $\mathsf C(U)$ for every
factorization chart $U$;
\item every structure map $f\colon U\to V$ carries a specified
equivalence
\[
 \beta_f\colon f_!b_U\xrightarrow{\ \sim\ }b_V;
\]
the maps $\beta_f$ satisfy the identity and composition laws
$\beta_{g\circ f}\simeq
\beta_g\circ g_!(\beta_f)$ together with their higher coherences;
\item every disjoint pair carries a specified equivalence
$\beta_{U,V}\colon
b_{U\sqcup V}\xrightarrow{\sim}b_U\boxtimes b_V$ under the factorization
equivalence
$\mathsf C(U\sqcup V)\simeq\mathsf C(U)\otimes\mathsf C(V)$;
the maps $\beta_{U,V}$ satisfy the unital, symmetric, and associative
factorization coherences and commute with the maps $\beta_f$;
\item the resulting K\"unneth map
\[
 \RHom_{\mathsf C(U)}(b_U,b_U)\otimes
 \RHom_{\mathsf C(V)}(b_V,b_V)
 \xrightarrow{\ \sim\ }
 \RHom_{\mathsf C(U\sqcup V)}(b_{U\sqcup V},b_{U\sqcup V})
\]
is an equivalence.
\end{enumerate}
Set
\[
 A_b(U):=\RHom_{\mathsf C(U)}(b_U,b_U).
\]
The corestriction maps and the K\"unneth equivalences make $A_b$ a
factorization $E_1$-algebra.  The compact-opposite level-$0$ category
is
\[
 \cC^\op(U):=\bigl(\mathsf C(U)^\omega\bigr)^\op .
\]
\end{definition}

\begin{definition}[Primitive open chart datum;
\ClaimStatusDefinitional]
\label{def:mr-open-datum}
\index{open chart datum|textbf}
A \emph{primitive open chart datum} on a tangential log curve is a
tuple
\[
 \mathfrak D_{\mathrm{op}}
 =
 (X,D,\tau;\mathsf C,b,\varepsilon_b),
\]
with the following entries.
\begin{enumerate}[label=\textup{(\roman*)}]
\item $(X,D,\tau)$ is a smooth projective curve over a
characteristic-zero field, equipped with a reduced divisor and
tangential structure.
\item $\mathsf C$ is a stable presentable open-sector factorization
category on $\Ran^{\mathrm{oc}}(X,D,\tau)$.
\item $b=(b_U,\beta_f,\beta_{U,V})$ is a factorization compact
generator in the sense of
Definition~\ref{def:mr-factorization-compact-generator}.
\item The associated factorization $E_1$-algebra is
\[
 A_b(U):=\RHom_{\mathsf C(U)}(b_U,b_U).
\]
\item The vacuum character is a morphism of unital factorization
$E_1$-algebras
\[
 \varepsilon_b\colon A_b\longrightarrow\mathbf1_{\fact}.
\]
Its chart maps $\varepsilon_{b,U}\colon A_b(U)\to k$ commute with
corestriction and the factorization K\"unneth equivalences.
\end{enumerate}
The category of primitive open chart data is denoted
$\mathsf{OpenDatum}(X,D,\tau)$.
\end{definition}

 B_{\varepsilon_b}(A_b)

 \cC^\op=(\mathsf C^\omega)^\op.

\end{definition}

\begin{remark}[Geometric and chart presentations;
\ClaimStatusProvedHere]

\begin{definition}[Dependency, vacuum, level, and quadrant discipline;
\ClaimStatusDefinitional]
\label{def:mr-dependency-vacuum-level-quadrant-discipline}
\index{dependency graph!typed arrows}
\index{chosen vacuum!Morita chart}
\index{level alphabet!Open quadrant}
\index{quadrant grid!Open and CY}
In this chapter a \emph{dependency arrow} records a proved implication

The chosen vacuum datum is the pair $(\mathsf C,b)$ over the

$\cC^\op=(\mathsf C^\omega)^\op$; level~1 is

 \mathsf C\simeq\mathrm{RMod}_{A_b}^{\fact},
 \qquad
 \cC^\op\simeq\mathrm{Perf}(A_b)^\op .
\]
The category $\cC^\op$ and the algebra $A_b$ thus occupy the

Theorem~\ref{thm:morita-invariance-HH}.

The level alphabet is fixed once for the Open quadrant:

$(\mathrm{Open}\cup\mathrm{CY})\times(\mathrm{Cat}\cup

\ClaimStatusConjectured.
\end{definition}

\section{The six-node open Beilinson reconstruction fork}
\label{sec:mr-tower}

The primitive datum produces two reconstruction equivalences and
three further constructions.  A modular enhancement enters through
genus-zero restriction and admits a separate scalar projection.

\begin{definition}[Open Beilinson reconstruction fork on
$(X,D,\tau)$; \ClaimStatusDefinitional]
\label{def:mr-tower}
\index{Beilinson tower!open quadrant}
The \emph{open Beilinson reconstruction fork} on $(X,D,\tau)$ is the
typed diagram
\[
\begin{tikzcd}[column sep=3.2em,row sep=2.3em]
&& B_{\varepsilon_b}(A_b)
   \arrow[dl,bend right=13,"\Omegach_X"']
\\
(\mathsf C,b)
   \arrow[r,bend left=13,"F_0"]
& (A_b,\varepsilon_b)
   \arrow[l,bend left=13,"\mathrm{Morita}"]
   \arrow[ur,bend left=13,"\Bbarch_X"]
   \arrow[d,"\mathfrak Z_b"]
   \arrow[dr,"\mathfrak O_b"]
\\
& Z_{\ch}^{\mathrm{chart}}(A_b)
& \mathfrak O_b
\\
&& \mathfrak F
   \arrow[u,"\mathrm{res}_0+\rho"']
   \arrow[r,"\pi_{\mathrm{scal}}"]
& \bigl(\mathfrak O^K_{g,n},
          \operatorname{obs}^{\mathrm{Hdg}}_{g,n},
          F_{g,n}(-;\alpha)\bigr).
\end{tikzcd}
\]
Here the six levels are: the factorization category at level~$0$; its
pointed Morita chart at level~$1$; the full bar coalgebra at level~$2$;
the chart cochain center at level~$3$; the operator package
$\mathfrak O_b$ at level~$4$; and the category-valued modular functor
$\mathfrak F$ together with its scalar projection at level~$5$.
The package $\mathfrak O_b$ comprises the intrinsic endofunctor line
category, the pointed diagonal-bimodule datum, and the
quadratic--Verdier label category introduced in
Definition~\ref{def:mr-pointed-operator-data}.  The horizontal and
upper arrows become equivalences under $H_0$ and $H_1$,
respectively.  The arrows $\mathfrak Z_b$ and $\mathfrak O_b$ are
functorial constructions.  The lower arrows are genus-zero restriction followed
by a specified realization $\rho$, and scalar projection.
\end{definition}

\section{Reconstruction and formation theorems with named hypothesis
packages}
\label{sec:mr-reconstruction}
%===========================================================

The open Beilinson fork is governed by two reconstruction
equivalences, three functorial constructions, and their comparison
theorems.  Every statement displays its hypothesis package.

\begin{theorem}[$F_0$-reconstruction: factorization Morita;
\ClaimStatusProvedHere]
\label{thm:mr-morita}
\index{reconstruction!Morita; level 0 to 1}
\textup{[Type: Open, Cat, $0\leftrightarrow 1$, hypothesis package
$H_0$.]}
Let $U\mapsto\mathsf C(U)$ be a stable presentable open-sector
factorization category on $(X,D,\tau)$, and let $b=(b_U)$ be a
factorization compact generator.  For every factorization chart $U$,
put $A_b(U)=\RHom_{\mathsf C(U)}(b_U,b_U)$.  The Morita functors
\[
 \Psi_U\colon\mathsf C(U)\longrightarrow
 \mathrm{RMod}_{A_b(U)},
 \qquad c\longmapsto\RHom_{\mathsf C(U)}(b_U,c),
\]
assemble into equivalences of factorization categories
\[
 \mathsf C\xrightarrow{\ \sim\ }
 \mathrm{RMod}^{\fact}_{A_b},
 \qquad
 \cC^\op=(\mathsf C^\omega)^\op
 \xrightarrow{\ \sim\ }
 \mathrm{Perf}(A_b)^\op .
\]

\textbf{Hypothesis package $H_0$.} The structure functors of
$\mathsf C$ are exact and colimit preserving; they preserve compact
objects; $b$ satisfies clauses \textup{(i)--(iv)} of
Definition~\ref{def:mr-factorization-compact-generator}; and the
factorization transition maps are enriched over $\Ch(k)$.
\end{theorem}

\begin{proof}
The presentable part of
Theorem~\ref{prop:morita-equivalence-compact-gen} gives
$\Psi_U\colon\mathsf C(U)\simeq\mathrm{RMod}_{A_b(U)}$ on every
factorization chart.  Let $f\colon U\to V$ be a structure map.  The
enriched functor $f_!$ and the chosen equivalence
$\beta_f\colon f_!b_U\simeq b_V$ induce the algebra morphism
\[
 f_b\colon A_b(U)\longrightarrow A_b(V)
\]
and, naturally in $c\in\mathsf C(U)$, the comparison
\[
 \alpha_f(c)\colon
 \RHom_{\mathsf C(U)}(b_U,c)
 \otimes^{\mathbf L}_{A_b(U)}A_b(V)
 \longrightarrow
 \RHom_{\mathsf C(V)}(b_V,f_!c).
\]
To\"en's derived Eilenberg--Watts theorem
\cite[Theorem~7.2 and Corollary~7.6]{Toen07} represents the composite
$\Psi_Vf_!\Psi_U^{-1}$ by the bimodule
\[
 \Psi_V(f_!b_U)
 \xrightarrow[\beta_f]{\ \sim\ }
 \Psi_V(b_V)=A_b(V),
\]
whose left $A_b(U)$-action is $f_b$.  Hence $\alpha_f$ is an
equivalence.  Equivalently, both sides preserve colimits in $c$, and
$\alpha_f(b_U)$ is the displayed equivalence on the compact generator.
The coherences of the maps $\beta_f$ identify the composites
$\alpha_{g\circ f}$ and $\alpha_g\alpha_f$.  Clause~\textup{(iv)} and
the maps $\beta_{U,V}$ identify the disjoint-union comparisons with
the factorization product of $A_b$.  The functors $\Psi_U$ therefore
assemble into an equivalence of factorization cosheaves.  Compact
preservation gives
$\mathsf C^\omega\simeq\mathrm{Perf}(A_b)$; passage to opposites gives
the level-$0$ equivalence.
\end{proof}

\begin{theorem}[$F_1$-reconstruction: associative bar--cobar
equivalence, Theorem~A;
\ClaimStatusConditional]
\label{thm:mr-A}
\index{reconstruction!Theorem A; level 1 to 2}
\textup{[Type: Open, Chain, $1\leftrightarrow 2$, pro-nilpotent

$H_1=H_{\mathrm{fact}}\cup H_{\mathrm{conv}}$ of

The bar--cobar adjunction

 \Omegach_X\;:\;
 \mathsf{CoAlg}^{\fact,\mathrm{dp}}_{\operatorname{Bar}(\mathsf{Ass}),X}
 \rightleftarrows
 \mathsf{Alg}^{\fact,\aug}_{\mathsf{Ass},X}\;:\;\Bbarch_X
\]
is an equivalence.  Its counit and unit are
\[
 \Omegach_X\Bbarch_X(A)\xrightarrow{\ \sim\ }A,
 \qquad
 C\xrightarrow{\ \sim\ }\Bbarch_X\Omegach_X(C).
\]
\textbf{Hypothesis package $H_1$.}
The reduced associative operad lies in the pro-nilpotent chiral Ran
ambient of Francis--Gaitsgory.  $H_{\mathrm{fact}}$ supplies the

bar and cobar under disjoint-locus factorization, collision descent,
units, and completion.  $H_{\mathrm{conv}}$ realizes the completed
chain models through exhaustive complete filtrations.
\end{theorem}

\begin{proof}

Theorem~\ref{thm:koszul-reflection} give the enhanced equivalence in
the Ran ambient.  Clause~\ref{KR-iii} restricts it under
$H_{\mathrm{fact}}$.
\end{proof}

\begin{theorem}[Quadratic Koszul recognition, Theorem~B;
\ClaimStatusConditional]
\label{thm:mr-B}
\index{recognition!Theorem B; quadratic level 1 to 2}
\textup{[Type: Open, Chain, $1\leftrightarrow2$, chosen quadratic
presentation, hypothesis package
$H_{\mathrm{CL}}(A,A^i,\tau_i)$.]}
Let $A=T(V)/(R)$ be a complete augmented quadratically presented
chiral algebra, let
\[
 A^i=C(s^{-1}V,s^{-2}R),
 \qquad
 q_A\colon A^i\longrightarrow B_X(A)
\]
be the quadratic coalgebra and its canonical comparison with the full
bar coalgebra.  Under $H_{\mathrm{CL}}(A,A^i,\tau_i)$, the following conditions
are equivalent:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $q_A$ is a quasi-isomorphism of complete chiral coalgebras;
\item the adjoint map $\Omega_X(A^i)\to A$ is a quasi-isomorphism;
\item the left and right twisted tensor products for the quadratic
twisting morphism $\tau_i\colon A^i\to A$ are acyclic.
\end{enumerate}
Thus Theorem~B recognizes the quadratic Koszul locus of the chosen
presentation.
\end{theorem}

\begin{proof}
This is Theorem~\ref{thm:bar-cobar-inversion-qi}, the chiral transport
of the Loday--Vallette quadratic twisting criterion
\cite[Theorems~2.3.2 and~3.4.6]{LV12}.  The package
$H_{\mathrm{CL}}(A,A^i,\tau_i)$ supplies factorization descent,
finite-window comparison, and strong convergence.
\end{proof}

\begin{proposition}[Fixed-coalgebra second-kind comparison;
\ClaimStatusConditional]
\label{prop:mr-fixed-C-second-kind}
\index{Positselski!fixed-coalgebra lane}
\textup{[Type: Open, Chain, level-$2$ module categories, fixed chiral

Let $C$ be a fixed complete chiral CDG-coalgebra satisfying

 D^{\mathrm{co}}(C\text{-}\mathrm{CoFact})

\]
An algebra comparison from a twisting morphism

Theorem~A carries the universal bar--cobar unit and counit; Theorem~B

fixed-coalgebra second-kind equivalence.
\end{proposition}

\begin{proof}
Apply Theorem~\ref{thm:off-koszul-ran-inversion} to the fixed

\end{proof}

\begin{theorem}[Level-$3$ formation and family-indexed support,
Theorem~H; \ClaimStatusConditional]
\label{thm:mr-H}
\index{formation!Theorem H; level 1 to 3}
\textup{[Type: Open, Chain, $1\rightarrow 3$, hypothesis packages
$H_Z$ and $H_H(A;S)$.]}
The level-$3$ operation forms the chart cochain center
\[
 Z_{\ch}^{\mathrm{chart}}(A)
 =C^\bullet_{\ch}(A,A).
\]
Fix a finite set $S\subset\mathbb Z$.  The family support package
$H_H(A;S)$ consists of a complete filtered cochain model $Q_A$, a
quasi-isomorphism $Q_A\xrightarrow{\sim}C^\bullet_{\ch}(A,A)$, a
cochain complex $K_{A,S}$ supported in $S$, and maps
\[
 K_{A,S}\mathop{\xrightarrow{\ \iota\ }}
 Q_A\mathop{\xrightarrow{\ p\ }}K_{A,S},
 \qquad p\iota=\mathrm{id},
\]
together with a filtration-continuous homotopy $h$ satisfying
\[
 dh+hd=\mathrm{id}_{Q_A}-\iota p.
\]
The package also includes convergence of the filtration and, when
$Q_A$ is first constructed on ordered configurations, a
quasi-isomorphism from its homotopy descent to the symmetric chiral
complex.  Under $H_H(A;S)$ the inclusion $\iota$ induces
\[
 H^n(K_{A,S})\xrightarrow{\ \sim\ }
 \ChirHoch^n(A)
 \qquad(n\in\mathbb Z),
\]
so the cohomology has support $S$.

The support is an invariant of the algebra and coefficient theory.
For bounded vertex-algebra cohomology, Bakalov--De Sole--Kac compute
the rank-one even free superboson as
\[
 \dim H^n=(2,1,0,0,\ldots)
\]
and the Virasoro vertex algebra as one-dimensional in degrees
$0,2,3$ and supported on those degrees
\cite[Theorems~7.4 and~7.2]{BDSK21}.  Transport of these benchmarks to
$C^\bullet_{\ch}(A,A)$ uses a specified comparison quasi-isomorphism
between the two cochain theories.
\end{theorem}

\begin{proof}
The displayed homotopy makes $(\iota,p,h)$ a strong deformation
retract.  Hence $\iota$ is a quasi-isomorphism.  Filtered convergence
and the ordered-to-symmetric comparison transport this
quasi-isomorphism to $C^\bullet_{\ch}(A,A)$.  The two benchmark vectors
are the cited computations of Bakalov--De Sole--Kac.
\end{proof}

\begin{definition}[Intrinsic lines, chart center, and Koszul-dual
labels; \ClaimStatusDefinitional]
\label{def:mr-pointed-operator-data}
Let $(\mathsf C,b)$ be a pointed factorization category and put
$A=A_b$.  The intrinsic line category is the monoidal category

\begin{proposition}[Multiplicity in a scalar fibre;
\ClaimStatusProvedHere]
\label{prop:mr-scalar-fibre-multiplicity}

$\mathbb Z/2$ fusion rules, the same nondegenerate $S$-matrix, the
same quantum dimensions, and the same closed Verlinde dimensions
$2^g$.  Their twists are
\[
 T_+=\operatorname{diag}(1,i),
 \qquad
 T_-=\operatorname{diag}(1,-i).
\]
Consequently the fibre of the closed-scalar projection contains at
least two balanced braided equivalence classes.  Category-valued
sewing retains the twist and recovers the genus-zero datum through
Theorem~\ref{thm:mr-modular}.
\end{proposition}

\begin{proof}
Both categories have simple objects $\mathbf1,s$ with
$s\otimes s\simeq\mathbf1$ and quantum dimensions $(1,1)$.  Their
$S$-matrix is the rank-two Fourier matrix.  The Verlinde formula
therefore gives $2^g$ on a closed genus-$g$ surface in both cases.
The twist of the nontrivial simple object is respectively $i$ and
$-i$; a balanced braided equivalence preserves this eigenvalue.
\end{proof}

\section{The Master Reconstruction and Formation Theorem}

The open quadrant contains two reconstruction equivalences and three

\begin{theorem}[Master Reconstruction and Formation Theorem on the
open quadrant]
\label{thm:mr-master}
\ClaimStatusConditional
\index{Master Reconstruction Theorem!open quadrant}
\index{open Beilinson tower!six-node reconstruction fork}
\textup{[Type: Open, Cat$\,\&$Chain, $0\rightarrow 5$, hypothesis
package
$H_0\cup H_1\cup H_Z\cup H_{\mathrm{CL}}\cup
H_{\mathrm{VD}}\cup H_{\mathrm{line}}\cup H_5\cup

$A_b=\RHom_{\mathsf C}(b,b)$.  Fix a factorization augmentation
$\varepsilon_b\colon A_b\to\mathbf1$.  Let $\mathfrak F$ be a
connected category-valued modular functor.  These data instantiate
the reconstruction fork of Definition~\ref{def:mr-tower} as follows.
\begin{enumerate}[label=\textup{(M\arabic*)}]
\item \label{M1}
\emph{Chart-side reconstruction.}

 \mathsf C\simeq\mathrm{RMod}^{\fact}_{A_b},

 A_b^i\xrightarrow{\sim}B_X(A_b),
 \qquad

 \rho_{A_b}^!\colon\mathsf{Line}^{!}(A_b)

\]
\end{enumerate}
\end{theorem}

\begin{proof}

\ref{M2} is Propositions~\ref{prop:mr-factorization-ew-lines}

\ref{M3} combines Theorem~\ref{thm:mr-B} and clause~\ref{KR-v} of

Theorem~\ref{thm:mr-drinfeld-double}.

\end{proof}
