\chapter{The Universal Conductor: ordered-to-symmetric averaging}
\label{chap:universal-conductor}

\epigraph{The conductor maps ordered data to its symmetric image and
records the homotopy fibre of the passage.}{}

The universal conductor of Volume~I is the averaging morphism
\[
 K_\cA \;=\; \Av_\cA \colon \gAE \longrightarrow \gAmod
\]
from the ordered convolution dg Lie algebra to the symmetric modular
quotient. At chain level it is the coinvariant quotient, with the
Reynolds idempotent furnishing its characteristic-zero splitting,
\[
 B^{\Ord}(\cA)=T^c(s^{-1}\bar\cA)
 \longrightarrow
 B^{\Sigma}(\cA)=\Symm^c(s^{-1}\bar\cA),
\]
and at the factorisation level it is left Kan extension along the
quotient map
\[
 q \colon \Ranord(X) \longrightarrow \Ran(X)=\Ranord(X)/\Sigma_\bullet.
\]
This is the map that sends the ordered bar package to its modular
shadow. Bar--cobar inversion, Verdier Koszul duality, the derived
chiral centre $\DerZ(\cA)$, and the categorical Drinfeld centre
$\mathcal Z(\Rep^{E_1}(\cA))$ are different constructions. The five objects
\[
 \cA,\qquad B(\cA),\qquad \cA^i,\qquad \cA^!,\qquad \DerZ(\cA)
\]
remain distinct throughout.

On standard witness lanes three scalar packages are kept separate from
the averaging morphism:
\[
 K^c(A,A^!)=c(A)+c(A^!),\qquad
 K^\kappa(A,A^!)=\kappa(A)+\kappa(A^!),\qquad
 K_{\mathrm{ghost}}^{\mathrm{leg}}(A)
 =-c_{\mathrm{ghost}}(\BRST_{\mathrm{can}}(A)).
\]
They are central-charge, modular-characteristic, and BRST-presentation
numbers, respectively.

$K$ has named chain-level and $(\infty,1)$-categorical witnesses. In
the untwisted characteristic-zero window its image is simultaneously a
coinvariant quotient, a Reynolds fixed-point summand, and a
homotopy-coinvariant factorisation shadow. In the general braided
window it is the $L_R$-twisted derived coinvariant descent of the
ordered bar. Its formal homotopy fibre is the ordered
information-loss conductor; it contains the ordered classes that
separate the four archetypes $G/L/C/M$. The landscape map
$\mathfrak U\colon \cA\mapsto\{G,L,C,M\}$ is the supremal
$G/L/C/M$ depth of the averaged shadow, checked row-by-row against the
census. The Vol~III
comparison is restricted to the ordered Koszul locus. The scalar
packages $K^c$, $K^\kappa$, and $K_{\mathrm{ghost}}^{\mathrm{leg}}$
are family-dependent and remain separate unless a displayed
calculation identifies them.

\section{Definition of the averaging conductor}
\label{sec:uc-definition}

An ordered chiral algebra $\cA$ on $X$ is
\emph{conductor-eligible} if its ordered bar package satisfies the
following hypothesis package
$\mathbf H_{\mathrm{uc}}(\cA)$:
\begin{enumerate}[label=\textup{(R\arabic*)}]
\item the ordered arity-$n$ bar carrier is finite type or complete
separated conilpotent in every arity, so finite group averaging and
derived coinvariants commute with the inverse-limit filtration;
\item the ordered differential, residue operations, and convolution
bracket are equivariant for the chosen $\Sigma_\bullet$-descent datum;
\item in the braided case, the meromorphic $R$-matrix satisfies the
quantum Yang--Baxter equation and strong unitarity
$R^{21}(-z)R^{12}(z)=\id$;
\item the KZ/Cherednik local system $L_R$ determined by~$R$ is
$\Sigma_n$-equivariant on $\Conf_n^{\ord}(X)$ and has regular-singular
nearby-cycle extension to the relevant Fulton--MacPherson collision
faces;
\item the nearby cycles of $L_R$ are compatible with the chiral
collision products used in the ordered bar differential.
\item in every finite arity window where a single symmetric dg~Lie
target is asserted, the local-system fibres \(R_n=(L_R)_{x_n}\) carry
shuffle-compatible \(\Sigma_m\times\Sigma_n\)-equivariant
coefficient-multiplication maps
\[
\mu_{m,n}\colon
\operatorname{Ind}_{\Sigma_m\times\Sigma_n}^{\Sigma_{m+n}}
(R_m\otimes R_n)\longrightarrow R_{m+n}
\]
so that the tensor-product twist produced by the ordered convolution
bracket lands in the chosen arity-\((m+n)\) descent target.
\item the symmetric bar target carries the shuffle/factorization
coproduct obtained from finite direct image along
$\Conf_n^{\ord}(X)\to\Conf_n(X)$, and the comparison with the
aritywise coinvariant carrier intertwines this coproduct with collision
factorization.  This is the coalgebra-upgrade datum.
\end{enumerate}
The displayed Reynolds formulas in this chapter use the untwisted case
$R=\mathrm{id}$. In the twisted case every occurrence of ordinary
$\Sigma_n$-coinvariants means the corresponding
$L_R$-twisted derived completed coinvariants.  The package
$\mathbf H_{\mathrm{uc}}(\cA)$ is precisely the domain on which the
aritywise coinvariant descent carries the dg Lie and
shuffle/factorization coalgebra structures.  The Reynolds fixed-point
model receives the transported structures; broader aritywise windows
retain its linear-shadow interpretation.

\subsection{Chain-level witness}
\label{subsec:uc-chain-level}

\begin{definition}[Reynolds--coinvariant averaging;
\ClaimStatusDefinitional]
\label{def:uc-reynolds-coinvariant}
Let $V=s^{-1}\bar\cA$. For each $n\geq 0$, let
\[
 \Av^{\mathrm{Rey}}_{\Sigma_n}
 \;:=\;
 \frac{1}{n!}\sum_{\sigma\in\Sigma_n}\sigma
 \colon
 V^{\otimes n}\longrightarrow V^{\otimes n}
\]
be the Reynolds projector and let
\[
 q_n \colon V^{\otimes n}\longrightarrow V^{\otimes n}_{\Sigma_n}
 \cong \Symm^n(V)
\]
be the coinvariant quotient. The \emph{chain-level universal
conductor} on the underlying arity-graded bar object is
\[
 K_{\cA}^{\mathrm{ch}}
 \;:=\;
 \bigoplus_{n\geq 0} q_n\circ \Av^{\mathrm{Rey}}_{\Sigma_n}
 \colon
 B^{\Ord}(\cA)=T^c(V)\longrightarrow B^{\Sigma}(\cA)=\Symm^c(V).
\]
When the ordered differential is $\Sigma_\bullet$-equivariant this is a
chain map to the symmetric coinvariant quotient.  Since
$q_n(\sigma v)=q_n(v)$,
\begin{equation}
\label{eq:uc-q-reynolds-is-q}
q_n\circ\Av^{\mathrm{Rey}}_{\Sigma_n}=q_n.
\end{equation}
Thus $q_n$ is the descent map, while the Reynolds idempotent supplies
its characteristic-zero fixed-point representative.  The notation
$\Symm^c(V)$ in this chapter carries the shuffle/factorization
coproduct specified in the final clause of
$\mathbf H_{\mathrm{uc}}(\cA)$.

The first coalgebra calculation fixes this convention.  For linearly
independent $v,w\in V$, put
\[
a=[v|w]-[w|v].
\]
Then $K^{\mathrm{ch}}_{\cA,2}(a)=0$, while the reduced tensor
deconcatenation gives
\[
(K^{\mathrm{ch}}_{\cA,1}\otimes K^{\mathrm{ch}}_{\cA,1})
\,\bar\Delta_T(a)=v\otimes w-w\otimes v,
\]
a nonzero tensor.  Hence raw tensor deconcatenation has a different
target type.  The coalgebra statement uses the
shuffle/factorization coproduct supplied by the hypothesis package.

In the $R$-twisted conductor domain, the preceding display is the
untwisted model for the same descent. The chain-level target is
\[
 B^{\Sigma}_{X,n}(\cA)
 \;\simeq\;
 \bigl(B^{\Ord}_{X,n}(\cA)\otimes L_R\bigr)_{\Sigma_n}^{\mathbb L,\wedge},
\]
where $L_R$ is the $\Sigma_n$-equivariant local system fixed in
$\mathbf H_{\mathrm{uc}}(\cA)$ and the superscript denotes derived
completed coinvariants. The \emph{ordered information-loss conductor}
in arity~$n$ is the homotopy fibre
\[
 \mathcal I_{\ord\to\Sigma,n}(\cA)
 :=
 \operatorname{Fib}\!\left(
 B^{\Ord}_{X,n}(\cA)\otimes L_R
 \longrightarrow
 B^{\Sigma}_{X,n}(\cA)
 \right).
\]
In the untwisted semisimple finite-arity window, Reynolds splitting
identifies this homotopy fibre with the completed kernel of
$q_n\circ\Av^{\mathrm{Rey}}_{\Sigma_n}$.
\end{definition}

\begin{definition}[Ordered-to-symmetric chiral homology descent;
\ClaimStatusDefinitional]
\label{def:uc-totalization}
For a smooth complex curve $X$, write
\[
 \int_X^{\Ord}\cA
 :=
 \Tot\!\Bigl(\bigoplus_{n\geq 0}
 C_\bullet(\Conf_n^{\Ord}(X),\cA^{\boxtimes n})\Bigr)
\]
for ordered chiral homology and
\[
 \int_X^{\mathrm{mod}}\cA
 :=
 \Tot\!\Bigl(\bigoplus_{n\geq 0}
 C_\bullet(\Conf_n(X),(\cA^{\boxtimes n})_{\Sigma_n})\Bigr)
\]
for the symmetric modular shadow. The induced descent map
\[
 K^{\mathrm{hom}}_{X,\cA}\colon
 \int_X^{\Ord}\cA\longrightarrow\int_X^{\mathrm{mod}}\cA
\]
for a conductor-eligible $\cA$ is defined on the $n$-th
totalisation summand by the same Reynolds
operator
$\frac{1}{n!}\sum_{\sigma\in\Sigma_n}\sigma$ followed by passage to
coinvariants. This is the totalisation-level realisation of the
ordered-to-symmetric descent of
\eqref{eq:ordered-descent}.
For nontrivial $R$, replace the summand by
$C_\bullet(\Conf_n^{\Ord}(X),\cA^{\boxtimes n}\otimes L_R)$ and take
the $L_R$-twisted derived completed $\Sigma_n$-coinvariant quotient.
\end{definition}

\subsection{$(\infty,1)$-categorical witness}
\label{subsec:uc-infinity-level}

\begin{definition}[Homotopy-coinvariant factorisation functor;
\ClaimStatusDefinitional]
\label{def:uc-homotopy-coinvariants}
Let
\[
 q\colon \Ranord(X)\longrightarrow \Ran(X)=\Ranord(X)/\Sigma_\bullet
\]
be the quotient map of \eqref{eq:ran-as-quotient}. For an ordered
factorisation algebra $\mathcal F$ on $\Ranord(X)$ with the descent
datum above, define
\[
 K^\infty(\mathcal F)
 \;:=\;
 q_!(\mathcal F)
 \;\simeq\;
 \mathcal F_{h\Sigma_\bullet},
\]
the left Kan extension along $q$, equivalently the homotopy
coinvariants of the symmetric-group action. Its right adjoint is the
ordered pullback $q^*$, whose fibrewise description is given by
homotopy fixed points.
With nontrivial $R$, the same notation means
$q_!(\mathcal F\otimes L_R)$; the untwisted formula is the special case
$L_R=\mathbb Q_{\mathrm{triv}}$.
\end{definition}

\begin{theorem}[Finite-window $R$-twisted descent for the differential
and bracket]
\label{thm:uc-r-twisted-dg-lie-descent}
\ClaimStatusProvedHere
\index{R-twisted descent!bar differential and bracket}
\index{ordered-to-symmetric descent!dg Lie criterion}
Type signature: \textup{(}Open quadrant, finite-window
ordered-to-symmetric conductor presentation, Beilinson levels~$2$ and
$5$, hypothesis package: characteristic zero, complete separated
finite-window filtration, strong-unitary \(R\)-twisted
\(\Sigma_\bullet\)-descent, regular-singular FM extension, and the
shuffle-compatible coefficient multiplication of
\(\mathbf H_{\mathrm{uc}}\)\textup{)}.
Fix a finite arity/weight/filtration window and put
\[
C_n:=B^{\Ord}_{X,n}(\cA)_{\mathcal W},\qquad
C_n\simeq\varprojlim_N C_n/F^N C_n .
\]
Let \(R_n=(L_R)_{x_n}\) be the finite-dimensional right
\(\mathbb Q[\Sigma_n]\)-module determined by the \(R\)-twisted
local-system fibre in that window, and define the completed twisted
coinvariant bar piece
\[
B^{\Sigma,R}_{X,n}(\cA;\mathcal W)
:=
\varprojlim_N
\left(
R_n\otimes_{\mathbb Q[\Sigma_n]} C_n/F^N C_n
\right).
\]
Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \textup{(bar differential)}
If \(d_{B,n}\) is equivariant for the \(R\)-twisted
\(\Sigma_n\)-descent datum, then it induces a continuous differential
on \(B^{\Sigma,R}_{X,n}(\cA;\mathcal W)\).  Thus averaging commutes
with \(d_B\) only under the stated equivariance.
\item \textup{(convolution bracket)}
Suppose the ordered convolution bracket
\[
[-,-]_{m,n}\colon
C_m\widehat\otimes C_n\longrightarrow C_{m+n}
\]
is equivariant for the block inclusion
\(\Sigma_m\times\Sigma_n\subset\Sigma_{m+n}\), and suppose the
coefficient maps \(\mu_{m,n}\) of
\(\mathbf H_{\mathrm{uc}}\) are supplied.  Then the bracket descends to
a continuous chain map
\[
[-,-]^{R}_{m,n}\colon
B^{\Sigma,R}_{X,m}(\cA;\mathcal W)
\widehat\otimes
B^{\Sigma,R}_{X,n}(\cA;\mathcal W)
\longrightarrow
B^{\Sigma,R}_{X,m+n}(\cA;\mathcal W).
\]
The coefficient map~\(\mu_{m,n}\) carries the natural tensor-product
twist
\(\operatorname{Ind}_{\Sigma_m\times\Sigma_n}^{\Sigma_{m+n}}
(R_m\otimes R_n)\) into the fixed \(R_{m+n}\)-twisted symmetric
bar piece.
\item \textup{(dg Lie morphism criterion)}
The descended ordered-to-symmetric map is a dg~Lie morphism exactly on
the finite windows satisfying \textup{(i)} and \textup{(ii)}.  In the
untwisted Reynolds model \(R_n=\mathbb Q_{\mathrm{triv}}\), the
Reynolds representative
\(R_{\Sigma_n}\colon C_n\to C_n^{\Sigma_n}\) is a dg~Lie morphism
precisely when
\[
R_{\Sigma_{m+n}}([u,v])=0
\quad
\text{for all }
u\in\ker R_{\Sigma_m},\ v\in\ker R_{\Sigma_n},
\]
equivalently when the Reynolds kernel is closed under the ordered
bracket in the relevant arities.  The differential equivariance and
bracket condition therefore mark the domain of the dg~Lie morphism;
the same formula on a broader window is a continuous linear projection
to a shadow.
\end{enumerate}
\end{theorem}

\begin{proof}
Part~\textup{(i)} is the defining quotient calculation for twisted
coinvariants.  The relations in
\(R_n\otimes_{\mathbb Q[\Sigma_n]} C_n/F^NC_n\) are
\[
rg\otimes c-r\otimes gc,\qquad g\in\Sigma_n .
\]
If \(d_{B,n}(gc)=g\,d_{B,n}(c)\), then
\[
d(rg\otimes c-r\otimes gc)
=rg\otimes d c-r\otimes g d c,
\]
so the differential preserves the relations on each finite quotient
and hence on the completed inverse limit.  This is the
bar-differential part of
Lemma~\ref{lem:R-twisted-completed-coinvariants}.

For~\textup{(ii)}, the same lemma identifies the completed twisted
coinvariant with the equivariant-map model
\(\operatorname{Hom}_{\Sigma_n}(R_n^\vee,C_n)\).  The ordered bracket
of equivariant maps
\(R_m^\vee\to C_m\) and \(R_n^\vee\to C_n\) is equivariant for
\(\Sigma_m\times\Sigma_n\), hence gives a map with coefficient
twist \(R_m\otimes R_n\).  Inducing from
\(\Sigma_m\times\Sigma_n\) to \(\Sigma_{m+n}\) records the shuffle
summation in the convolution bracket.  Composing with
\(\mu_{m,n}\) is exactly the datum that lands in the chosen
\(R_{m+n}\)-twisted target.  The tensor-product twist is the preceding
coefficient object in this construction.

Part~\textup{(iii)} is the comparison between the quotient model and
the Reynolds representative.  In the untwisted characteristic-zero
window, Lemma~\ref{lem:completed-reynolds-descent-ordered-arity}
identifies completed coinvariants with invariants by the Reynolds
idempotent.  Writing \(x=x_0+u\) and \(y=y_0+v\), with
\(x_0,y_0\) invariant and \(u,v\) in the Reynolds kernels, one has
\[
R_{\Sigma_{m+n}}([x,y])
=[x_0,y_0]+R_{\Sigma_{m+n}}([u,v]).
\]
The mixed terms vanish by equivariance.  Therefore
\(R([x,y])=[R x,R y]\) for all \(x,y\) exactly when the displayed
kernel condition holds.  Combining this with part~\textup{(i)} gives
the stated dg~Lie criterion and the final linear-shadow warning.
\end{proof}

\begin{theorem}[\textbf{Universal conductor as ordered-to-symmetric descent}]
\label{thm:uc-universal-conductor}
\ClaimStatusProvedHere
Type signature: \textup{(}Open quadrant, ordered-to-symmetric
conductor presentation, Beilinson levels~$2$ and~$5$, hypothesis
package $\mathbf H_{\mathrm{uc}}(\cA)$ above\textup{)}.
For every conductor-eligible ordered chiral algebra $\cA$ on a smooth
complex curve $X$,
the assignment
\[
 K_\cA=\Av_\cA\colon \gAE\longrightarrow \gAmod
\]
is a well-defined natural dg Lie morphism with the following two named
witnesses.
\begin{enumerate}
\item[\textup{(i)}] \emph{Chain-level witness.}
 $K_\cA$ is dual to the coinvariant map
 $K_{\cA}^{\mathrm{ch}}=q_\bullet$ of
 Definition~\ref{def:uc-reynolds-coinvariant} on
 $B^{\Ord}(\cA)\to B^{\Sigma}(\cA)$, equivalently to the
 totalisation-level descent
 $K_{X,\cA}^{\mathrm{hom}}$ of
 Definition~\ref{def:uc-totalization}.  The identity
 $q_\bullet\circ\Av^{\mathrm{Rey}}_{\Sigma_\bullet}=q_\bullet$
 supplies the fixed-point representative.
\item[\textup{(ii)}] \emph{$(\infty,1)$-categorical witness.}
 $K_\cA$ is induced by the homotopy-coinvariant functor
 $q_!\simeq(-)_{h\Sigma_\bullet}$ of
 Definition~\ref{def:uc-homotopy-coinvariants}.
\end{enumerate}
It satisfies:
\begin{enumerate}
\item[\textup{(a)}] $K_\cA$ preserves the dg Lie differential and the
 bracket because these operations are $\Sigma_\bullet$-equivariant on
 the conductor domain.
\item[\textup{(b)}] $K_\cA$ is lossy on the nontrivial
 $\Sigma_\bullet$-isotypic ordered sector: it remembers the symmetric
 modular shadow. The discarded ordered commutator, associator, and
 higher braid data form the formal conductor
 $\mathcal I_{\ord\to\Sigma,\bullet}(\cA)$, the literal homotopy fibre
 of the passage.
\item[\textup{(c)}] $K_\cA$ is distinct from
 $\Omega(B(\cA))\to \cA$, from the bar-dual coalgebra
 $\cA^i=C_X(s^{-1}V,s^{-2}R)$ and its comparison
 $q_\cA\colon\cA^i\to B^{\mathrm{ch}}(\cA)$, from the Verdier algebra
 $K_X(\cA)=\mathbb D_{\Ran}B^{\mathrm{ch}}(\cA)$ and its pair map
 $\nu_\cA\colon K_X(\cA)\to\cA^!$, from the derived centre
 $\DerZ(\cA)$, and from the categorical Drinfeld centre
 $\mathcal Z(\Rep^{E_1}(\cA))$.
\end{enumerate}
\end{theorem}

\begin{proof}
The ordered bar differential and the ordered convolution bracket are
defined by sums over splittings and collision residues. On the
conductor domain these operations are equivariant for the specified
$\Sigma_\bullet$-descent datum; in the untwisted symmetric case this is
the usual compatibility of the collision forms with relabelling,
together with the Arnold relation in the Orlik--Solomon algebra. Hence
the Reynolds projector commutes with the differential, and the quotient
$q_n\colon V^{\otimes n}\to \Symm^n(V)$ is a chain map.  Equivariance
is the condition that upgrades the aritywise linear projection to a dg
Lie morphism. This proves the chain-level part.

In the $R$-twisted case, the quantum Yang--Baxter equation supplies the
triple-overlap cocycle for the elementary transpositions and strong
unitarity gives the Coxeter involutivity relation
$R^{21}(-z)R^{12}(z)=\id$. The regular-singular
Fulton--MacPherson extension hypothesis gives nearby cycles of
$L_R$ along collision faces, and the compatibility clause in
$\mathbf H_{\mathrm{uc}}(\cA)$ identifies those nearby cycles with the
chiral collision products in the ordered bar differential. Therefore
the same argument applies to
$B^{\Ord}_{X,n}(\cA)\otimes L_R$ after replacing ordinary
coinvariants by derived completed $L_R$-twisted coinvariants, as in
Theorem~\ref{thm:uc-r-twisted-dg-lie-descent},
Corollary~\ref{cor:eight-cor-R-descent} and
Lemma~\ref{lem:R-twisted-completed-coinvariants}.

The factorisation-theoretic part is the ordered-to-symmetric descent
already used in the ordered modular package: since
$\Ran(X)=\Ranord(X)/\Sigma_\bullet$, left Kan extension along $q$
identifies the symmetric object with fibrewise homotopy coinvariants,
or with fibrewise $L_R$-twisted homotopy coinvariants in the braided
case.
Passing to the dual convolution algebras gives the dg Lie morphism
$K_\cA\colon \gAE\to \gAmod$.

Finally, the source and target objects distinguish $K$ from the other
load-bearing constructions. The bar--cobar counit
$\Omega(B(\cA))\to \cA$ is an inversion map; the bar-dual coalgebra is
$\cA^i=C_X(s^{-1}V,s^{-2}R)$, equipped with
$q_\cA\colon\cA^i\to B^{\mathrm{ch}}(\cA)$; its Koszul recognition
statement is the quasi-isomorphism of~$q_\cA$.  Verdier duality forms
$K_X(\cA)=\mathbb D_{\Ran}B^{\mathrm{ch}}(\cA)$, and the chosen
Koszul-pair map $\nu_\cA\colon K_X(\cA)\to\cA^!$ reaches the dual
algebra.  The object $\DerZ(\cA)$ is the algebraic
closed-sector Hochschild cochain algebra; and
$\mathcal Z(\Rep^{E_1}(\cA))$ is a braided monoidal category. The
universal conductor is the ordered-to-symmetric projection.
\end{proof}

\section{Image of the conductor}
\label{sec:uc-image}

\begin{theorem}[\textbf{Three descriptions of the image}]
\label{thm:uc-trinity}
\ClaimStatusProvedHere
Let $V=s^{-1}\bar\cA$, work over characteristic zero, restrict to the
conductor domain, and in the displayed formulas assume the untwisted
finite-dimensional arity window. For each $n\geq 0$, the image of the
universal conductor on $V^{\otimes n}$ is characterised by three
independent descriptions.
\begin{enumerate}
\item[\textup{(Q)}] \emph{Quotient description.}
 \[
 \operatorname{im}(K_{\cA,n}^{\mathrm{ch}})
 \;=\;
 V^{\otimes n}_{\Sigma_n}
 \;\cong\;
 \Symm^n(V).
 \]
 Thus the image is, by construction, the symmetric coinvariant
 quotient.
\item[\textup{(F)}] \emph{Fixed-point splitting.}
 The Reynolds projector identifies the same image with the fixed-point
 summand
 \[
 \operatorname{im}(\Av^{\mathrm{Rey}}_{\Sigma_n})
 =
 (V^{\otimes n})^{\Sigma_n}
 \cong
 \Symm^n(V).
 \]
 Thus the characteristic-zero Reynolds splitting represents the image
 as a fixed-point object.
\item[\textup{(H)}] \emph{Homotopy-coinvariant description.}
 On factorisation algebras the same image is the essential image of
 \[
 q_!\simeq(-)_{h\Sigma_\bullet}
 \colon
 \Fact(\Ranord(X))\longrightarrow \Fact(\Ran(X)).
 \]
 Hence the image is the homotopy-coinvariant shadow of the ordered
 theory.
\end{enumerate}
Consequently the answer to ``subalgebra, fixed point, or quotient?''
is precise: the image is a coinvariant quotient by definition, a
fixed-point summand after Reynolds splitting, and a sub-dg Lie algebra
of the symmetric target~$\gAmod$.  The fixed-point summand receives the
operations transported from the quotient.  For fixed-point
representatives $x=\mathrm{Re}_m\widetilde x$ and
$y=\mathrm{Re}_n\widetilde y$, the transported bracket is
\[
[x,y]_{\mathrm{tr}}
=\mathrm{Re}_{m+n}[\widetilde x,\widetilde y].
\]
It agrees with the restriction of the ordered bracket precisely under
the Reynolds-kernel criterion of
Theorem~\ref{thm:uc-r-twisted-dg-lie-descent}.
For nontrivial $R$, replace these symmetric-power displays by the
$L_R$-twisted derived completed coinvariant image.  A chosen twisted
descent equivalence supplies any subsequent comparison with
$\Symm^n(V)$.
\end{theorem}

\begin{proof}
Description~(Q) is the definition of
Definition~\ref{def:uc-reynolds-coinvariant}: the target of
$K_{\cA,n}^{\mathrm{ch}}$ is the coinvariant quotient.
Description~(F) is the characteristic-zero Reynolds splitting:
averaging projects $V^{\otimes n}$ onto its $\Sigma_n$-invariant
summand, and invariants are canonically identified with symmetric
tensors.  Equation~\eqref{eq:uc-q-reynolds-is-q} identifies this as a
linear splitting of the coinvariant quotient; transport supplies its
dg Lie operations. Description~(H) is the factorisation-theoretic reformulation
of the quotient map $\Ranord(X)\to \Ran(X)$ from
Definition~\ref{def:uc-homotopy-coinvariants}. The final sentence just
unpacks the three descriptions.
\end{proof}

\begin{proposition}[Schur--Weyl kernel count]
\label{prop:uc-kernel-dimension}
\ClaimStatusProvedHere
On an untwisted finite-dimensional purely even arity window with
$d=\dim V$, one
has
\[
 \dim\ker(K_{\cA,n}^{\mathrm{ch}})
 \;=\;
 d^n-\binom{n+d-1}{d-1}.
\]
Equivalently, the image has dimension $\binom{n+d-1}{d-1}$.
For a graded or super window this binomial coefficient is replaced by
the coefficient of the graded-symmetric Hilbert series.
\end{proposition}

\begin{proof}
This is the explicit kernel count of
Proposition~\ref{prop:ker-av-schur-weyl}: the image of averaging on
$V^{\otimes n}$ is $\Symm^n(V)$ and stars-and-bars gives its
dimension. Subtracting from $d^n$ yields the kernel dimension.
\end{proof}

\begin{theorem}[Degree-two conductor kernel and ordered
\texorpdfstring{$r$}{r}-matrix data]
\label{thm:uc-degree-two-rmatrix-kernel}
\ClaimStatusProvedHere
\index{averaging map!degree-two kernel}
\index{ordered r-matrix!kernel of averaging}
\index{braid data!higher averaging kernel}
Type signature: \textup{(}Open quadrant, finite-window
ordered-to-symmetric conductor presentation, Beilinson levels~$2$ and
$5$, hypothesis package: characteristic zero, completed Reynolds
descent of Theorem~\ref{thm:uc-r-twisted-dg-lie-descent}, and the

\end{proof}

\section{Central-charge, \texorpdfstring{$\kappa$}{kappa}, and ghost scalar packages}

The averaging conductor $K_\cA=\Av_\cA$ is a morphism of dg Lie
algebras.  Scalar packages arise after further functors and choices.
A central-charge companion sum receives two conformal vectors.  A
modular-characteristic sum also receives a normalization of~$\kappa$.
A ghost scalar receives a concrete conformal BRST presentation,
including its charged fields, neutral fields, and improvement term.
The notation $A^!=(A^i)^\vee$ receives its algebra-level meaning on the
finite-type or continuously completed Verdier lane.

\begin{definition}[Three scalar conductor packages; \ClaimStatusDefinitional]
\label{def:uc-scalar-packages}
For a specified conformal companion pair $(A,A^!)$, set
\[
 K^c(A,A^!) := c(A)+c(A^!),\qquad
 K^\kappa(A,A^!) := \kappa(A)+\kappa(A^!).

For a chosen finite conformal BRST presentation~$\mathcal P$, set
\[
 K_{\mathrm{ghost}}^{\mathrm{leg}}(\mathcal P)
 :=
 -\,c_{\mathrm{ghost}}(\mathcal P).
\]
The first number is central-charge complementarity; the second is
modular-characteristic complementarity; the third is a
presentation-sensitive ghost central-charge scalar.
\end{definition}
