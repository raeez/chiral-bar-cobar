\chapter{$W$-algebras: local structure, bar transport, and scalar
companions}
\label{chap:w-algebra-koszul}
\label{chap:w-algebras}

A $W$-algebra enters the Open Beilinson tower through four distinct
layers.  The local layer is its vertex-algebra OPE\@.  The filtered layer
is the Li--Kazhdan degeneration to the arc space of a Slodowy slice.
The Koszul layer is the completed chiral bar coalgebra and its Verdier
dual.  The modular layer is the genus-one trace produced after sewing.
The first two layers supply the exact calculations in this chapter.
The final two carry named hypothesis packages whenever the required
comparison theorem remains open.

This separation is especially important for two scalar families.  For
principal $\mathcal W_N$, the central-charge companion sum is an exact
rational-function identity.  For Bershadsky--Polyakov, the standard
Fehily--Kawasetsu--Ridout conformal vector gives conductor~$50$; a
shifted conformal vector gives the secondary sum~$196$.  The modular
characteristic of Bershadsky--Polyakov remains a genus-one
calculation.

\section{Drinfeld--Sokolov reduction and Slodowy geometry}

\begin{definition}[Slodowy transverse slice]
\label{def:slodowy-slice}
Let $(e,h,f)$ be an $\mathfrak{sl}_2$-triple in a complex semisimple
Lie algebra $\mathfrak g$.  The Slodowy slice through $f$ is
\begin{equation}\label{eq:slodowy-slice-repaired}
S_f:=f+\mathfrak g^e,
\qquad
\mathfrak g^e=\ker(\operatorname{ad}e).
\end{equation}
It is an affine space transverse to the orbit of $f$ at the point
$f$ and carries the Poisson structure induced by Hamiltonian
reduction.
\end{definition}

\begin{remark}[Arc-space PBW theorem]
\label{rem:jet-slodowy}
For the universal affine $W$-algebra, the Li--Kazhdan PBW theorem is
the differential-commutative identification
\begin{equation}\label{eq:jet-slodowy-repaired}
\operatorname{gr}_{\mathrm{Li}}\mathcal W^k(\mathfrak g,f)
\cong \mathbb C[J_\infty S_f].
\end{equation}
Since $S_f=f+\mathfrak g^e$ is affine space,
$\mathbb C[J_\infty S_f]$ is a free differential polynomial algebra
on $(\mathfrak g^e)^*$.  This PBW statement concerns the associated
graded vertex algebra; a completed-bar collapse requires the separate
criterion in Theorem~\ref{thm:w-pbw-slodowy-collapse}.
\end{remark}

\begin{theorem}[Associated varieties under DS reduction;
\ClaimStatusProvedElsewhere]
\label{thm:arakawa-variety-intersection}
Let $M$ lie in a category on which the higher DS cohomology vanishes.
Arakawa's associated-variety theorem gives
\begin{equation}\label{eq:arakawa-variety-repaired}
X_{H^0_{\mathrm{DS},f}(M)}=X_M\cap S_f
\end{equation}
with the scheme-theoretic formulation and finiteness hypotheses of
\cite{Ara15,ArakawaMoreau17}.
\end{theorem}

\begin{remark}[Geometric localization]
\label{rem:ds-geometric-localization-principle}
Drinfeld--Sokolov reduction restricts the classical support from the
ambient current space to the transverse slice.  Equation
\eqref{eq:arakawa-variety-repaired} is the support-theoretic form of
this localization.  Transport of a chiral bar coalgebra is a stronger
derived assertion: it asks for a filtered comparison between the bar
construction of the BRST resolution and the bar construction of its
DS cohomology.
\end{remark}

\begin{remark}[The PBW face of the reconstruction square]
\label{rem:pbw-slodowy-master-square}
The PBW face compares
$\operatorname{gr}_{\mathrm{Li}}\mathcal W^k(\mathfrak g,f)$ with
$\mathbb C[J_\infty S_f]$.  The derived face compares completed bar
coalgebras.  A theorem joining these faces consists of a convergent
filtered spectral sequence, diagonal concentration, and a continuous
twisting morphism.  These are precisely the hypotheses recorded
below.
\end{remark}

\section{Exact principal type-$A$ arithmetic}

For $N\ge2$, put
\[
\mathcal W_N^k:=\mathcal W^k(\mathfrak{sl}_N,f_{\mathrm{prin}}),
\qquad t:=k+N.
\]
The standard DS central charge is
\begin{equation}\label{eq:principal-WN-central-charge}
c_N(k)
=(N-1)-\frac{N(N^2-1)(k+N-1)^2}{k+N}.
\end{equation}

\begin{proposition}[$\mathcal W_3$ central charge;
\ClaimStatusProvedHere]
\label{prop:w3-central-charge}
For $N=3$,
\begin{equation}\label{eq:w3-central-charge-repaired}
c_3(k)=2-\frac{24(k+2)^2}{k+3}.
\end{equation}
The companion involution $k^\vee=-k-6$ satisfies
\begin{equation}\label{eq:w3-central-sum}
c_3(k)+c_3(k^\vee)=100.
\end{equation}
\end{proposition}

\begin{proof}
Equation~\eqref{eq:w3-central-charge-repaired} is
\eqref{eq:principal-WN-central-charge} at $N=3$.  Writing $t=k+3$
gives
$c_3(k)=2-24(t-1)^2/t$ and
$c_3(k^\vee)=2+24(t+1)^2/t$; their sum is~$100$.
\end{proof}

\begin{definition}[Zamolodchikov $\mathcal W_3$ algebra]
\label{def:w3-algebra}
The universal $\mathcal W_3$ algebra over $\mathbb C(c)$ is generated
by $T$ and $W$ of weights $2$ and $3$, with
\begin{align}
T(z)T(w)&\sim \frac{c/2}{(z-w)^4}
+\frac{2T(w)}{(z-w)^2}+\frac{\partial T(w)}{z-w},\nonumber\\
T(z)W(w)&\sim \frac{3W(w)}{(z-w)^2}
+\frac{\partial W(w)}{z-w},\nonumber\\
W(z)W(w)&\sim \frac{c/3}{(z-w)^6}
+\frac{2T(w)}{(z-w)^4}+\frac{\partial T(w)}{(z-w)^3}\nonumber\\
&\quad+\frac{\frac3{10}\partial^2T(w)
+\frac{32}{22+5c}\Lambda(w)}{(z-w)^2}
+\frac{\frac1{15}\partial^3T(w)
+\frac{16}{22+5c}\partial\Lambda(w)}{z-w},
\label{eq:w3-ope-repaired}
\end{align}
where
$\Lambda={:}TT{:}-\frac3{10}\partial^2T$.
Theorem~\ref{thm:w-w-ope-complete} gives the full normalization proof
and the OPE-to-mode conversion.
\end{definition}

\begin{proposition}[Principal $W_3/W_N$ standard-family ledger;
\ClaimStatusConditional]
\label{prop:principal-WN-standard-family-ledger}
The exact local and arithmetic data are:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $\mathcal W_N^k$ has principal strong-generator weights
$2,3,\ldots,N$ on the generic PBW lane.
\item The companion level $k^\vee=-k-2N$ obeys
\begin{equation}\label{eq:principal-WN-conductor}
K_N^c:=c_N(k)+c_N(k^\vee)=4N^3-2N-2.
\end{equation}
\item The same integer is the principal ghost spin sum
\begin{equation}\label{eq:principal-WN-ghost-sum}
\sum_{s=2}^N2(6s^2-6s+1)=4N^3-2N-2.
\end{equation}
\item Choose the diagonal generator normalization
\begin{equation}\label{eq:wn-2pt}
W^{(s)}(z)W^{(s)}(w)\sim
\frac{c/s}{(z-w)^{2s}}+\cdots.
\end{equation}
Under the genus-one diagonal-trace hypothesis
$H_{\mathrm{diag}}^{g=1}$, the modular characteristic is
\begin{equation}\label{eq:wn-kappa}
\kappa(\mathcal W_N^k)=c_N(k)(H_N-1),
\qquad H_N=\sum_{j=1}^N\frac1j.
\end{equation}
\end{enumerate}
Thus $K_3^c=100$.  Under $H_{\mathrm{diag}}^{g=1}$ and the
DS/bar companion package,
$K_3^\kappa=(H_3-1)K_3^c=250/3$.
\end{proposition}

\begin{proof}
Write $A=N(N^2-1)$.  In the coordinate $t=k+N$,
\[
c_N(k)=N-1-A\frac{(t-1)^2}{t},\qquad
c_N(k^\vee)=N-1+A\frac{(t+1)^2}{t}.
\]
Addition gives $2(N-1)+4A=4N^3-2N-2$.  The finite sums
$\sum s$ and $\sum s^2$ give
\eqref{eq:principal-WN-ghost-sum}.  The final clause is the stated
diagonal genus-one hypothesis applied to~\eqref{eq:wn-2pt}.
\end{proof}

\ClaimStatusConditional]
\label{thm:wn-obstruction}
\textbf{Type signature.}

\]

\end{theorem}

\section{Bershadsky--Polyakov: standard and shifted lanes}

Let
$\mathcal B^k=\mathcal W^k(\mathfrak{sl}_3,f_{(2,1)})$.
Fehily--Kawasetsu--Ridout use an ordinary vertex algebra whose strong
generators $J,G^+,G^-,T$ are all even, of respective weights
$1,3/2,3/2,2$~\cite[Definition~2.1]{FKR20}.  The standard central
charge is
\begin{equation}\label{eq:bp-standard-central-repaired}
c_{\mathrm{BP}}(k)
=-\frac{(2k+3)(3k+1)}{k+3}
=25-6(k+3)-\frac{24}{k+3}.
\end{equation}
Therefore
\begin{equation}\label{eq:bp-standard-conductor-repaired}
c_{\mathrm{BP}}(k)+c_{\mathrm{BP}}(-k-6)=50.
\end{equation}
The explicitly shifted secondary convention is
\begin{equation}\label{eq:bp-shifted-central-repaired}
c_{\mathrm{BP}}^{\mathrm{shift}}(k)
=2-\frac{24(k+1)^2}{k+3},
\qquad
c_{\mathrm{BP}}^{\mathrm{shift}}(k)
+c_{\mathrm{BP}}^{\mathrm{shift}}(-k-6)=196.
\end{equation}
The all-even reciprocal-weight diagnostic is
\begin{equation}\label{eq:bp-reciprocal-weight-diagnostic}
1+\frac{1}{3/2}+\frac{1}{3/2}+\frac12=\frac{17}{6}.
\end{equation}
It is a generator-weight diagnostic.  The genus-one curvature
$\kappa_{\mathrm{BP}}(k)$, its ratio to $c_{\mathrm{BP}}(k)$, and its
companion sum all have status \ClaimStatusOpen.

\ClaimStatusConditional]
\label{thm:w-bp-strict}

\begin{theorem}[Subregular Appell identity;
\ClaimStatusProvedHere]
\label{thm:w-subregular-appell}
Let $u_j=h+v_j$ for $0\le j<n$, with $\sum_jv_j=0$, and let
$e_a(v)$ be the elementary symmetric functions.  For $0\le r\le n$,
\begin{equation}\label{eq:subregular-appell-repaired}
\frac{1}{(n-r)!}\partial_h^{n-r}\prod_{j=0}^{n-1}u_j
=\sum_{t=0}^r\binom{n-r+t}{t}e_{r-t}(v)h^t.
\end{equation}
\end{theorem}

\begin{proof}
Expand $\prod_j(h+v_j)=\sum_{a=0}^ne_{n-a}(v)h^a$ and differentiate
term by term.
\end{proof}

\begin{theorem}[Subregular completed-dual classification;
\ClaimStatusConditional]
\label{thm:w-subregular-classification}
Assume the Feigin--Semikhatov Bell packet identifies the degree-$r$
term in~\eqref{eq:subregular-appell-repaired} with the arity-$r$
Taylor coefficient of the continuously dualized bar normal form, and
assume the completed PBW criterion of
Theorem~\ref{thm:w-pbw-slodowy-collapse}.  Then
$\mathcal W_n^{(2)}$ has canonical Taylor degree $n-1$.  The case
$n=3$ is the quadratic BP presentation of
Theorem~\ref{thm:w-bp-strict}.
\end{theorem}

\begin{theorem}[Filtered PBW bar-collapse criterion;
\ClaimStatusProvedHere]
\label{thm:w-pbw-slodowy-collapse}
Let $A$ be a complete filtered augmented chiral algebra.  Assume:
\begin{enumerate}[label=\textup{(\roman*)}]
\item the completed bar filtration is exhaustive, separated, and
weightwise complete;
\item its spectral sequence converges strongly;
\item the $E_1$ page is the bar homology of
$\operatorname{gr}A\cong\operatorname{Sym}_\partial(V)$ and is
concentrated on one bar/PBW diagonal.
\end{enumerate}
Then all higher differentials vanish for bidegree reasons, and the
completed bar homology is the completed exterior coalgebra on the
suspended differential generators.  The canonical twisting morphism
is a filtered quasi-isomorphism precisely when the resulting
extension data are complete.
\end{theorem}
