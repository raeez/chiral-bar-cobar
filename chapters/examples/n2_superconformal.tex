distinction between OPE data, a coset presentation, and a modular
characteristic.  Its Lie conformal algebra is explicit.  Its unitary
minimal models admit the Kazama--Suzuki realization.  A genus-one
curvature comparison for that coset is a further theorem.

\[
 (\text{Open quadrant},\ \text{$\mathcal N=2$ OPE/coset presentation},\

\section{OPE data and generators}

Let $V_c^{\mathcal N=2}$ denote the universal Neveu--Schwarz vertex
superalgebra at central charge $c$.  It is strongly generated by
\begin{equation}\label{eq:n2-generators}
 T\quad(h=2,\ \bar0),\qquad
 J\quad(h=1,\ \bar0),\qquad
 G^\pm\quad(h=\tfrac32,\ \bar1).
\end{equation}
In the convention of Kac~\cite{Kac98}, the singular OPEs are
\begin{align}
T(z)T(w)&\sim \frac{c/2}{(z-w)^4}
 +\frac{2T(w)}{(z-w)^2}+\frac{\partial T(w)}{z-w},
\label{eq:n2-ope-TT}\\
T(z)J(w)&\sim \frac{J(w)}{(z-w)^2}+\frac{\partial J(w)}{z-w},
\label{eq:n2-ope-TJ}\\
T(z)G^\pm(w)&\sim \frac{\frac32G^\pm(w)}{(z-w)^2}
 +\frac{\partial G^\pm(w)}{z-w},
\label{eq:n2-ope-TG}\\
J(z)J(w)&\sim \frac{c/3}{(z-w)^2},
\label{eq:n2-ope-JJ}\\
J(z)G^\pm(w)&\sim \frac{\pm G^\pm(w)}{z-w},
\label{eq:n2-ope-JG}\\
G^+(z)G^-(w)&\sim \frac{2c/3}{(z-w)^3}
 +\frac{2J(w)}{(z-w)^2}
 +\frac{2T(w)+\partial J(w)}{z-w},
\label{eq:n2-ope-GG}\\
G^\pm(z)G^\pm(w)&\sim 0.
\label{eq:n2-ope-self}
\end{align}
The opposite mixed OPE follows from vertex-superalgebra skew symmetry.

Integer spectral flow acts by the formulas in

\end{proposition}

\section{The Kazama--Suzuki presentation}

\begin{remark}[Kazama--Suzuki coset; \ClaimStatusProvedElsewhere]
\label{rem:n2-kazama-suzuki}
For $k\in\mathbb Z_{\ge0}$, the simple unitary $\mathcal N=2$ minimal
model is realized as the commutant of a diagonal Heisenberg algebra in
$L_k(\mathfrak{sl}_2)\otimes F$, where $F$ is a complex free-fermion
vertex superalgebra~\cite{KazamaSuzuki1989}.  Its central charge is
\begin{equation}\label{eq:n2-coset}
 c_{\mathcal N=2}(k)=\frac{3k}{k+2}.
\end{equation}
The generic universal coset comparison is denoted $H_{\mathrm{KS}}$;
it includes the chosen Heisenberg embedding, the commutant map, simple
quotient compatibility, and base change in $k$.
\end{remark}

The inverse parameter relation is
\[
 k=\frac{2c}{3-c}.
\]
Thus $c=3$ is the boundary value approached as $k$ tends to infinity,
while $k=-2$ is the pole of the rational parametrization.

\section{Modular characteristic}

\begin{definition}[Genus-one Kazama--Suzuki comparison package]
\label{def:n2-modular-comparison-package}
The package $H_{\mathcal N=2}^{\mathrm{mod}}$ consists of:
\begin{enumerate}[label=\textup{(\roman*)}]
\item chain models for the genus-one curvature of
 $V_k(\mathfrak{sl}_2)$, $F$, the diagonal Heisenberg algebra, and
 their commutant;
\item a trace-compatible exact coset triangle;
\item determinant-line and conformal-vector compatibility;
\item continuity through the completed commutant and simple quotient;
\item a comparison from the resulting curvature class to the scalar
 Hodge line.
\end{enumerate}
\end{definition}

\begin{proposition}[Conditional modular characteristic of the
$\mathcal N=2$ coset; \ClaimStatusConditional]
\label{prop:n2-kappa}
Assume $H_{\mathrm{KS}}$ and
$H_{\mathcal N=2}^{\mathrm{mod}}$.  Assume further that the three
constituent curvature classes have the common trace convention
\[
 \kappa\bigl(V_k(\mathfrak{sl}_2)\bigr)=\frac{3(k+2)}4,
 \qquad
 \kappa(F)=\frac12,
 \qquad
 \kappa(H_{\mathrm{diag}})=\frac{k}{2}+1.
\]
Then additivity in the trace-compatible coset triangle gives
\begin{equation}\label{eq:n2-sca-kappa}
 \kappa\bigl(V_c^{\mathcal N=2}\bigr)
 =\frac{k+4}{4}
 =\frac{6-c}{2(3-c)}.
\end{equation}
The pointed genus-one class satisfies
\[
 \operatorname{tr}_1
 \operatorname{Obs}^{\mathrm{def}}_{1,1}
 \bigl(V_c^{\mathcal N=2}\bigr)
 =\kappa\bigl(V_c^{\mathcal N=2}\bigr)\lambda_1.
\]
\end{proposition}

\begin{proof}
The assumed exact triangle gives
\[
 \kappa\bigl(V_c^{\mathcal N=2}\bigr)
 =\frac{3(k+2)}4+\frac12-\left(\frac{k}{2}+1\right)
 =\frac{k+4}{4}.
\]
Substitution of $k=2c/(3-c)$ gives the second expression.
\end{proof}

The proposition is an implication from a named genus-one package.
Constructing its trace-compatible coset triangle is the open
mathematical step.

\section{Parameter reflection and complementarity}

Define the critical affine reflection
\[
 \iota_{\mathrm{crit}}(k):=-k-4.
\]
Direct substitution into~\eqref{eq:n2-coset} gives
\[
 c_{\mathcal N=2}(k)+
 c_{\mathcal N=2}\bigl(\iota_{\mathrm{crit}}(k)\bigr)=6.
\]
This is an identity of rational functions.  A chiral Koszul
interpretation requires a Verdier--bar comparison between the two
levels.

\begin{proposition}[Conditional $\mathcal N=2$ complementarity;
\ClaimStatusConditional]
\label{prop:n2-complementarity}
Assume the hypotheses of Proposition~\ref{prop:n2-kappa} at $k$ and
$-k-4$, together with a duality comparison
\[
 H_{\mathcal N=2}^{\mathrm{KD}}\colon
 \mathbb D_{\operatorname{Ran}}B
 \bigl(V_{c(k)}^{\mathcal N=2}\bigr)
 \simeq V_{c(-k-4)}^{\mathcal N=2}.
\]
Then
\begin{equation}\label{eq:n2-sca-complementarity-sum}
 \kappa\bigl(c(k)\bigr)+\kappa\bigl(c(-k-4)\bigr)=1.
\end{equation}
\end{proposition}

\begin{proof}
The conditional formula~\eqref{eq:n2-sca-kappa} gives
\[
 \frac{k+4}{4}+\frac{(-k-4)+4}{4}=1.
\]
The package $H_{\mathcal N=2}^{\mathrm{KD}}$ supplies the object-level
meaning of the reflected second summand.
\end{proof}

The fixed point of $c\mapsto6-c$ is $c=3$.  It lies on the boundary
of the $k$-chart.  The fixed point of $k\mapsto-k-4$ is the pole
$k=-2$.  A completed comparison across this boundary is part of
$H_{\mathcal N=2}^{\mathrm{KD}}$.

\section{PBW data and the chiral bar problem}

\begin{definition}[$\mathcal N=2$ bar-collapse package]
\label{def:n2-bar-collapse-package}
The package $H_{\mathcal N=2}^{\mathrm{bar}}$ consists of a complete
Li filtration, convergence of the induced bar spectral sequence,
control of singular vectors in the selected quotient, and a
quasi-isomorphism from the quadratic coalgebra to the full bar
coalgebra.
\end{definition}

\begin{proposition}[PBW page and conditional quadratic recognition;
\ClaimStatusConditional]
\label{prop:n2-koszulness}
For the universal algebra $V_c^{\mathcal N=2}$, the Li-associated
graded is the differential supercommutative algebra generated by the
symbols of $J,T,G^+,G^-$ and their derivatives.  Consequently the Li
filtration determines the first page of the bar spectral sequence.
Under $H_{\mathcal N=2}^{\mathrm{bar}}$, the canonical comparison
\[
 q_{\mathcal N=2}\colon
 \bigl(V_c^{\mathcal N=2}\bigr)^{\mathrm i}
 \longrightarrow B\bigl(V_c^{\mathcal N=2}\bigr)
\]
is a quasi-isomorphism, and quadratic Koszul recognition follows.
\end{proposition}

\begin{proof}
The PBW assertion is the standard Li-filtration statement for a
freely generated universal vertex superalgebra.  The final implication
is precisely the quasi-isomorphism included in
$H_{\mathcal N=2}^{\mathrm{bar}}$ and the quadratic Comparison Lemma.
\end{proof}

\begin{remark}[The CE/chiral bar comparison]
\label{rem:n2-ce-chiral-gap}
The Chevalley--Eilenberg complex of the negative-mode Lie superalgebra
and the chiral bar complex have different inputs and filtrations.  A
comparison map between them is an additional datum.  The Li PBW page
provides the natural starting filtration for constructing that map.
\end{remark}

\section{MC3 status: coset descent}
\label{sec:n2-mc3-status}

\begin{conjecture}[MC3 through Kazama--Suzuki descent;
\ClaimStatusConjectured]
\label{conj:mc3-n2-sca-kazama-suzuki}
Suppose the ambient affine--fermion category is compactly generated,
the diagonal Heisenberg action is monadic, and commutant formation
preserves the selected compact generator.  Then the induced generator
of the $\mathcal N=2$ coset category is thick and the corresponding
Morita chart reconstructs the coset category.
\end{conjecture}

\begin{remark}[Coset descent obligation]
\label{rem:n2-mc3-coset-obstruction}
The required theorem is a Barr--Beck comparison for the diagonal
Heisenberg action, followed by compactness of the commutant generator.
The Kazama--Suzuki character identity supplies numerical evidence;
the monadic comparison supplies the categorical transition.
\end{remark}

\begin{remark}[Falsification test]
\label{rem:n2-mc3-falsification}
A decisive test computes the endomorphism algebra of the proposed
coset generator at the first nontrivial unitary levels $k=1,2$ and
compares its thick closure with the known finite module category.
Disagreement identifies the missing descent or compactness input.
\end{remark}

\begin{remark}[Alternative comparison routes]
\label{rem:n2-mc3-alternative-routes}
Adamovi\'c's lattice realization~\cite{Adamovic1999} and the
Feigin--Semikhatov realization~\cite{FeiginSemikhatov2004} provide two
candidate ambient categories.  Each route requires its own compact
generator, commutant comparison, and trace compatibility.
\end{remark}

\section{Spectral flow}
\label{sec:n2-spectral-flow}

For $\ell\in\mathbb Z$, choose the standard spectral-flow convention
\begin{equation}\label{eq:n2-spectral-flow}
\begin{aligned}
 \sigma^\ell(L_n)&=L_n+\ell J_n+\frac{c}{6}\ell^2\delta_{n,0},\\
 \sigma^\ell(J_n)&=J_n+\frac{c}{3}\ell\delta_{n,0},\\
 \sigma^\ell(G_r^\pm)&=G_{r\pm\ell}^\pm.
\end{aligned}
\end{equation}
Substitution in the Neveu--Schwarz commutation relations proves that
$\sigma^\ell$ is an automorphism of the mode algebra.  Transport of a
genus-one curvature class through spectral flow requires compatibility
of the determinant line and trace map; it belongs to
$H_{\mathcal N=2}^{\mathrm{mod}}$.
